*** Validation ELF_header
**** How works the header.
search man elf for the attributes of the structure:
=unsigned char e_ident[EI_NIDENT];= {Magicnbr, 32/64b, Dataencoding, version, OSABI, ABI, ...}
=uint16_t      e_type;= object file type {exec, rel, dyn, ...}
=uint16_t      e_machine;= required archictecture for a file *Ignored*
=uint32_t      e_version;= elf file version
=ElfN_Addr     e_entry;= entrypoint of the program (where is the first instruction to execute
=ElfN_Off      e_phoff;= Program header offset
=ElfN_Off      e_shoff;= Section header offset
=uint32_t      e_flags;= Processor-specific flags  *Ignored*
=uint16_t      e_ehsize;= elf header size Bytes
=uint16_t      e_phentsize;= elf program header size Bytes
=uint16_t      e_phnum;= nbr of program header
=uint16_t      e_shentsize;= elf section header size Bytes
=uint16_t      e_shnum;= nbr of section header
=uint16_t      e_shstrndx;= index of section header string (inside the array of section header)
***** Edge Cases
When the num of  =S/Phdr= is more than could be handle by uint16_t (65535) the first Sheader got the real info of =e_shnum, e_phnum, shstrndx=
*What value got then:*
- ===e_shnum == SHN_UNDEF===
  - ==e_shnum = shdr[0].sh_size== //Where to find
- ===e_phnum == PN_XNUM===
  - ==e_phnum = shdr[0].sh_info== //Where to find
- ===e_shstrndx == SHN_XINDEX===
  - ==e_shstrndx = shdr[0].sh_link== //Where to find

                          uint64_t size_bits)
  {
          uint64_t	limit;

          if (!times || !size)
                  return (false);
          if (!size_bits || size_bits > 64)
                  return (true);
          limit = 0;
          while (--size_bits)
                  limit = limit << 1 | 1;
          return (unsigned_overflow_limit(times, size, limit));
