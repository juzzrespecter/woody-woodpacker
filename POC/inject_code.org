         off_t 	file_size;

         if (argc == 1)
           return (BAD_NBR_PARAMS);
         fd = get_file (argv[1]);
         if (fd < 0)
           return (FAIL_OPEN_FD);
         if (get_file_size(fd, &file_size) != SUCCESS)
         {
             close (fd);
             return (FAIL_LSEEK);
         }
         file_mmap = mmap (NULL, file_size, PROT_NONE | PROT_READ,
                        NO_SHARE_CHANGES, fd, 0);
         close(fd);
         if (MAP-FAILED == file_mmap)
           return (FAIL_MMAP);
         ftype = get_ftype (file_mmap, &file_size);
         if (ftype == FT_UNKNOW)
           return (UNRECOGNIZED_FILE);
         return (SUCCESS);
     }

  int get_file_size(int fg, off_t *file_size)
  {
    *file_size = lseek(fd, 0, SEEK_END);
    if (*file_size == ERROR_STDL)
    {
        perror("Failed to seek the end of file");
        return (FAIL_LSEEK);
    }
    if (lseek(fd, 0, SEEK_SET) == ERROR_STDL)
    {
        perror("Failed to seek the begin of file");
        return (FAIL_LSEEK);
    }
    return (SUCCESS);
  }

#+begin_src c :tangle signature_check.c
     t_file_type get_ftype(const char *file_content, const off_t *file_size)
     {
       if (ELF_MAGIC_SIZE < *file_size && !strncmp(file_content, ELF_MAGIC, ELF_MAGIC_SIZE))
         return (FT_ELF);
       if (MACH_O_MAGIC_SIZE < *file_size && !strncmp(file_content, MACH_O_64b_MAGIC, MACH_O_MAGIC_SIZE))
         return (FT_MACH_O);
       if (MACH_O_MAGIC_SIZE < *file_size && !strncmp(file_content, MACH_O_64b_MAGIC, MACH_O_MAGIC_SIZE))
         return (FT_MACH_O);
       if (DOS_SIGN_SIZE < *file_size && !strncmp(file_content, DOS_SIGN_SIZE)
           return (FT_PE);
       return (FT_UNKNOW);
     }
