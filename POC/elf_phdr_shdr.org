  uint64_t elf_get_program_ndx_from_addr(t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t addr)
    {
      t_elf_ph phdr;
      uint64_t i;
      uint64_t p_vaddr;

      i = 0;
      while (++i < phnum) // Skip the first 
        {
          phdr = elf_get_program_header(ehdr, i);
          p_vaddr = elf_get_program_header_vaddr(&phdr, class);
  	//  is loadable phdr
          if (PT_LOAD == elf_get_program_header_type(&phdr, class) && is_between(p_vaddr, p_vaddr + elf_get_program_header_filesz(&phdr, class) - 1, addr))
            return  (i);
        }
      return  (i);
    }

  size_t elf_get_highest_loadable_vaddr_avalaible(t_elf_h *ehdr, unsigned char class, uint64_t phnum)
  {
    t_elf_ph phdr;
    uint64_t i;
    size_t p_vaddr;
    size_t highest_avalaible_addr = 0;

    i = 0;
    while (++i < phnum) // Skip the first 
      {
        phdr = elf_get_program_header(ehdr, i);
        if (elf_get_program_header_type(&phdr, class) != PT_LOAD) // If not loadable ignore
  	continue ;
        p_vaddr = elf_get_program_header_vaddr(&phdr, class);
        if (highest_avalaible_addr <= p_vaddr)
  	highest_avalaible_addr = get_alignment_effective_addr(elf_get_program_header_align(&phdr, class), p_vaddr + elf_get_program_header_memsz(&phdr, class));
      }
    return  (highest_avalaible_addr);
  }

    uint64_t elf_get_section_ndx_from_addr(t_elf_h *ehdr, size_t addr, unsigned char class, uint64_t shnum)
    {
      t_elf_sh		shdr;

        {
          shdr = elf_get_section_header(ehdr, i);

    }

  size_t ft_js(unsigned char *dst, size_t distance)
  {
    t_type hop = (t_type){.b8 = distance};
    size_t written;

    written = 0;
    memcpy(dst + written, "\x0f\x88", 2);
    written += 2;
    memcpy(dst + written, &hop.b4, sizeof(hop.b4));
    return (written + sizeof(hop.b4));
  }

  size_t ft_mov_asm_rsp(void *dst, size_t offset, size_t value) //mov the dst pointer to not overwrite.
  {
    size_t	written;
    const t_type	type = (t_type){.b8 = value};
    const t_type	off = (t_type){.b8 = offset};

    written = 0;
    if (dst)
      memcpy(dst, "\xc6", 1);
    written++;
    if (!off.b8)
      {
        if (dst)
          memcpy(dst + written, "\x04", 1);
      }
    else if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80) // if offset and byte syze
      {
        if (dst)
          memcpy(dst + written, "\x44", 1);
      }
    else // if offset and > byte size
      {
        if (dst)
          memcpy(dst + written, "\x84", 1);
      }
    written++;
    if (dst)
      memcpy(dst + written, "\x24", 1);
    written++;
    if (off.b8)
      {
        /* if (dst) */
        /* printf("off -> b value %u, b4 value %u, b8 value %lu\n", (uint32_t)off.b, off.b4, off.b8); */
        if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80)
          {
            if (dst)
              memcpy(dst + written, &off.b, sizeof (off.b)); 
            written += sizeof (off.b);
          }
        else if ((uint64_t)off.b4 == off.b8 && off.b4 < 0x80000000)
          {
            if (dst)
              memcpy(dst + written, &off.b4, sizeof (off.b4)); 
            written += sizeof (off.b4);
          }
        else
          {
            if (dst)
              memcpy(dst + written, &off.b8, sizeof (off.b8)); 
            written += sizeof (off.b8);
          }
      }
    if ((uint32_t)type.b  == type.b4 && (uint64_t)type.b4 == type.b8)
      {
        if (dst)
          memcpy(dst + written, &type.b, sizeof (type.b));
        written += sizeof (type.b);
      }
    else if ((uint64_t)type.b4 == type.b8 && type.b4)
      {
        if (dst)
          memcpy(dst + written, &type.b4, sizeof (type.b4));
        written += sizeof (type.b4);
      }
    else
      {
        if (dst)
          memcpy(dst + written, &type.b8, sizeof (type.b8));
        written += sizeof (type.b8);
      }
    return (written);
  }

  size_t ft_sub_asm_rsp_buffer_byte(void *dst, uint64_t offset, unsigned char value, unsigned char class)//mov the dst pointer to not overwrite.
   {
     size_t	written;
     const t_type	off = (t_type){.b8 = offset};

     written = 0;
     if (class == ELFCLASS64)
       {
         if (dst)
           memcpy(dst, "\x80", 1);
         written++;
       }

     if (!off.b8)
       {
         if (dst)
           memcpy(dst + written, "\x2c", 1);//sub byte
         written++;
       }
     else if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80)
       {
         if (dst)
           memcpy(dst + written, "\x6c", 1);//sub byte
         written++;
       }
     else 
       {
         if (dst)
           memcpy(dst + written, "\xac", 1);//sub byte
         written++;
       }
     if (dst)
       memcpy(dst + written, "\x24", 1);// rsp pointer
     written++;
     if (off.b8)
       {
         if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80)
           {
          if (dst)
            memcpy(dst + written, &off.b, sizeof(off.b));
          written += sizeof (off.b);
           }
         else
           {
             if (dst)
               memcpy(dst + written, &off.b4, sizeof(off.b4));
             written += sizeof (off.b4);
           }
       }
     if (dst)
       memcpy(dst + written, &value, sizeof (value));
     written += sizeof(value);
     return (written);
   }

    void set_jmp_entrypoint(void *payload, uint64_t entrypoint, unsigned char class)
    {
      t_elf_addr u_entrypoint = (t_elf_addr){.b64 = entrypoint};
      size_t offset = 0;

      if (class == ELFCLASS64)
        {
          memcpy(payload, "\x48", 1); //rex prefix x64
          offset++;
        }
      memcpy(payload + offset, "\xb8", 1); //mov eax in case preceding rex is rax.
      offset++;
      if (class == ELFCLASS64)
        {
          memcpy(payload + offset, &u_entrypoint.b64, sizeof (u_entrypoint.b64));
          offset+= sizeof (u_entrypoint.b64);
        }
      else
        {
          memcpy(payload + offset, &u_entrypoint.b32, sizeof (u_entrypoint.b32));
          offset+= sizeof (u_entrypoint.b32);
        }
      memcpy(payload + offset, "\xff\xe0", 2); 
    }

    void encrypt(unsigned char *src, size_t size)
    {
      size_t i = 0;
      while ( i < size)
          src[i++] += 13;
    }

  size_t decrypt_rot13_asm(unsigned char *dst, size_t payload_size, unsigned char class)
  {
    size_t written = 0;
    size_t i = 0;

    while (i < payload_size)
      { /* dst + written segfault */
        if (dst)
          written += ft_sub_asm_rsp_buffer_byte(dst + written, i, 13, class);
        else
          written += ft_sub_asm_rsp_buffer_byte(dst, i, 13, class);
        i++;
      }
    return (written);
  }

  static size_t copy_payload_to_mmap_aux(unsigned char *dst, size_t offset)
  {
    size_t written;
    written = ft_mov_cl_from_rsp(dst, offset);
    if (dst)
      return (written + ft_mov_rax_from_cl(dst + written, offset));
    return (written + ft_mov_rax_from_cl(dst, offset));
  }

  size_t copy_payload_to_mmap(unsigned char *dst, size_t payload_size)
  {
    size_t written = 0;
    size_t i = 0;

    while (i < payload_size)
      { /* dst + written segfault if dst == NULL*/
        if (dst)
          written += copy_payload_to_mmap_aux(dst + written, i);
        else
          written += copy_payload_to_mmap_aux(dst, i);
        i++;
      }
    return (written);
  }

  int generate_payload(void *elf_mmap, char **payload_ptr, size_t *payload_size)
  {
    // 1. Begin payload stuff
    // get the section to include the payload
    t_elf_h 		ehdr = (t_elf_h){.elf32 = elf_mmap};
    const uint64_t     e_entrypoint = elf_get_elfh_entry(&ehdr);
    /* const uint64_t 	shnum = elf_get_elfh_shnum(&ehdr); */
    const uint64_t 	phnum = elf_get_elfh_phnum(&ehdr);
    const unsigned char 	class = elf_get_elfh_e_ident_class(elf_mmap);
    // in case find the section show relevant info function.

    int cypher_payload_fd;
    if (class == ELFCLASS64)
      cypher_payload_fd = open("bytecode/payload_64.bin", O_RDONLY);
    else
      cypher_payload_fd = open("bytecode/payload_32.bin", O_RDONLY);
    if (cypher_payload_fd < 0)
      return (5);
    const off_t cypher_payload_fd_size = lseek(cypher_payload_fd,0, SEEK_END);
    if (cypher_payload_fd_size == -1)
      return (6);
    if (lseek(cypher_payload_fd, 0, SEEK_SET) == -1)
      return (7);
    off_t jmp_size = get_mov_rax_jmp_rax_size(class);

    const uint64_t cypher_payload_size = cypher_payload_fd_size + jmp_size;
    unsigned char *cypher_payload = malloc(cypher_payload_size);
    if (!cypher_payload)
      return (8);
    if (read(cypher_payload_fd, cypher_payload, cypher_payload_fd_size) != cypher_payload_fd_size)
      return (9);
    close(cypher_payload_fd);
    set_jmp_entrypoint(cypher_payload + cypher_payload_fd_size, e_entrypoint, class);
    encrypt(cypher_payload, cypher_payload_size);
    size_t bytecode_buffer_size = 0;
    size_t cypher_payload_index = 0;
    while (cypher_payload_index < cypher_payload_size)
      {
        bytecode_buffer_size += ft_mov_asm_rsp(NULL, cypher_payload_index, cypher_payload[cypher_payload_index]);
        cypher_payload_index++;
      }
    bytecode_buffer_size += decrypt_rot13_asm(NULL, cypher_payload_size, class);

    size_t size_of_cp_payload_to_mmap = copy_payload_to_mmap(NULL, cypher_payload_size); // + 4. 
    int bytecode_until_mmap_fd = open("bytecode/open_mmap_until_mmap_size.bin", O_RDONLY);
    if (bytecode_until_mmap_fd < 0)
      return (1);
    const off_t bytecode_until_mmap_fd_size = lseek(bytecode_until_mmap_fd,0, SEEK_END);
    if (bytecode_until_mmap_fd_size == -1)
      return (6);
    if (lseek(bytecode_until_mmap_fd, 0, SEEK_SET) == -1)
      return (7);
    int bytecode_after_mmap_fd = open("bytecode/open_mmap_after_mmap_addr.bin", O_RDONLY);
    if (bytecode_after_mmap_fd < 0)
      return (1);
    const off_t bytecode_after_mmap_fd_size = lseek(bytecode_after_mmap_fd,0, SEEK_END);
    if (bytecode_after_mmap_fd_size == -1)
      return (6);
    if (lseek(bytecode_after_mmap_fd, 0, SEEK_SET) == -1)
      return (7);

    uint32_t highest_loadable_vaddr;
    uint32_t mmap_size;
    size_t generated_bytecode_size = bytecode_buffer_size + size_of_cp_payload_to_mmap + 3 * ft_sub_asm_rsp(NULL, cypher_payload_size, class) + jmp_size + set_jmp_rax(NULL) + get_js_size() + bytecode_after_mmap_fd_size + bytecode_until_mmap_fd_size + set_mov_edi(NULL, 0) + sizeof(mmap_size);
    ;
    char *generated_bytecode = malloc(generated_bytecode_size);
    if (!generated_bytecode)
      return (10);
    size_t generated_bytecode_index = 0;
    generated_bytecode_index += ft_sub_asm_rsp(generated_bytecode, cypher_payload_size, class);
    cypher_payload_index = 0;
    while (cypher_payload_index < cypher_payload_size)
      {
        generated_bytecode_index += ft_mov_asm_rsp(generated_bytecode + generated_bytecode_index, cypher_payload_index, cypher_payload[cypher_payload_index]);
        cypher_payload_index++;
      }
    generated_bytecode_index += decrypt_rot13_asm((void *)generated_bytecode + generated_bytecode_index, cypher_payload_size, class);
    char *bytecode_until_mmap_buffer = malloc(bytecode_until_mmap_fd_size);
    if (!bytecode_until_mmap_buffer)
      return (8);
    if (read(bytecode_until_mmap_fd, bytecode_until_mmap_buffer, bytecode_until_mmap_fd_size) != bytecode_until_mmap_fd_size)
      return (9);
    close(bytecode_until_mmap_fd);

    memcpy(generated_bytecode + generated_bytecode_index, bytecode_until_mmap_buffer, bytecode_until_mmap_fd_size);
    generated_bytecode_index += bytecode_until_mmap_fd_size;
    mmap_size = get_alignment_effective_addr(0x1000, cypher_payload_size);
    memcpy(generated_bytecode + generated_bytecode_index, &mmap_size, sizeof(mmap_size));
    generated_bytecode_index += sizeof(mmap_size);
    highest_loadable_vaddr = elf_get_highest_loadable_vaddr_avalaible(&ehdr, class, phnum);

    generated_bytecode_index += set_mov_edi(generated_bytecode + generated_bytecode_index, highest_loadable_vaddr);
    char *bytecode_after_mmap_buffer = malloc(bytecode_after_mmap_fd_size);
    if (!bytecode_after_mmap_buffer)
      return (8);
    if (read(bytecode_after_mmap_fd, bytecode_after_mmap_buffer, bytecode_after_mmap_fd_size) != bytecode_after_mmap_fd_size)
      return (9);
    close(bytecode_after_mmap_fd);
    memcpy(generated_bytecode + generated_bytecode_index, bytecode_after_mmap_buffer, bytecode_after_mmap_fd_size);
    generated_bytecode_index += bytecode_after_mmap_fd_size;

    generated_bytecode_index += ft_js((void *)generated_bytecode + generated_bytecode_index, size_of_cp_payload_to_mmap + 4 + jmp_size); // jmp 4 cause the address need 4 bytes and jmp_size got a mov_rax_jmp_rax append at the end of mmap_copy
    // copy payload to mmap
    generated_bytecode_index += copy_payload_to_mmap((void *)generated_bytecode + generated_bytecode_index, cypher_payload_size);

    generated_bytecode_index += ft_add_asm_rsp(generated_bytecode + generated_bytecode_index, cypher_payload_index, class);
    memcpy(generated_bytecode + generated_bytecode_index, "\xff\xe0", 2); //jmp_rax
    generated_bytecode_index += 2;
    generated_bytecode_index += ft_add_asm_rsp(generated_bytecode + generated_bytecode_index, cypher_payload_index, class);
    set_jmp_entrypoint(generated_bytecode + generated_bytecode_index, e_entrypoint, class);
    generated_bytecode_index += jmp_size;
    if (generated_bytecode_index != generated_bytecode_size)
      printf("generated_bytecode_size - generated_bytecode_index = %lu -> reverse: %lu\n", generated_bytecode_size - generated_bytecode_index, generated_bytecode_index - generated_bytecode_size);

    if (write(2, generated_bytecode, generated_bytecode_size) != (ssize_t)generated_bytecode_size)
      printf("error on write the generated bytecode\n");
    ,*payload_size = generated_bytecode_size;
    ,*payload_ptr = generated_bytecode;
    return (0);
  }

  static int prepare_info(t_elf_h *ehdr, unsigned char class, uint64_t payload_size,
  			uint64_t *next_phdr_vaddr, uint64_t phnum, uint64_t *new_phdr_offset)
  {
      t_elf_ph phdr = elf_get_program_header(ehdr, phnum - 1, class);
      const uint64_t offset = elf_get_program_header_offset(&phdr, class);
      const uint64_t filesz = elf_get_program_header_filesz(&phdr, class);
      const uint64_t new_align = 0x1000;
      ,*next_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
      ,*next_phdr_offset = get_alignment_effective_addr(new_align, offset + filesz);

      if (*next_phdr_vaddr % new_align == next_phdr_offset % new_align) 
        printf("all fine with addr and offset aligned\n");
      else
        {
        printf("need to modify a little addr and offset aligned");
        if (*next_phdr_vaddr % new_align > *next_phdr_offset % new_align)
  	,*next_phdr_offset += *next_phdr_vaddr % new_align - *next_phdr_offset % new_align;
        else
  	,*next_phdr_vaddr += *next_phdr_offset % new_align - *next_phdr_vaddr % new_align;
        }
      return 0;
  }

  static int copy_and_modify_elf_header(t_elf_h *ehdr, int packed, uint64_t new_e_shoff, unsigned char class,
  				      uint64_t new_entrypoint, uint64_t phnum, uint64_t shnum, uint64_t *writed)
  {
    uint64_t elf_struct_size = elf_get_struct_size_elf_header((char *)ehdr->elf32);
    t_elf_h ehdr_cp = {.elf32 = malloc(elf_struct_size)};
    if (!ehdr_cp.elf32)
      return 14;
    memcpy(ehdr_cp.elf32, ehdr->elf32, elf_struct_size);
    if (new_next_offset)
      elf_ehdr_set_shoff(&ehdr_cp, new_e_shoff, class);

    elf_ehdr_set_phnum(&ehdr_cp, ++phnum, class);

  }

  static bool last_section_in_phdr(t_elf_h *ehdr, t_elf_ph ndx_ph, uint64_t ndx_sh, unsigned char class, uint64_t phnum)
  {
    t_elf_sh shdr = elf_get_section_header(ehdr, ++ndx_sh);

    return (elf_get_program_ndx_from_addr(ehdr, class, phnum, elf_get_section_header_addr(&shdr, class)) != ndx_ph);
  }

  static int write_program_headers(t_elf_h *ehdr, void *elf_mmap, int packed, uint64_t ndx_ph, uint64_t phnum,
                                   unsigned char class, uint64_t payload_size, uint64_t *writed, uint64_t ndx_sh)
  {
      uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
      ,*writed += write(packed, elf_mmap + *writed, phdr_struct_size * ndx_ph);

      t_elf_ph phdr_cp = {.elf32 = malloc(phdr_struct_size)};
      if (!phdr_cp.elf32)
          return 15;

      t_elf_ph phdr = elf_get_program_header(ehdr, ndx_ph);
      memcpy(phdr_cp.elf32, phdr.elf32, phdr_struct_size);

      uint64_t p_align = elf_get_program_header_align(&phdr, class);
      uint64_t p_memsz = elf_get_program_header_memsz(&phdr, class);
      uint64_t p_filesz = elf_get_program_header_filesz(&phdr, class);
      uint64_t p_offset = elf_get_program_header_offset(&phdr, class);

      uint64_t p_og_padding = get_alignment_effective_addr(p_align, p_offset + p_filesz) - p_offset;
      uint64_t p_modified_padding = get_alignment_effective_addr(p_align, p_offset + p_filesz + payload_size) - p_offset;

      if (last_section_in_phdr(&ehdr, ndx_ph, ndx_sh, class, phnum)) // Modify phdr content.
        {
  	elf_phdr_set_filesz(&phdr_cp, p_filesz + payload_size, class);
  	elf_phdr_set_memsz(&phdr_cp, p_memsz + payload_size, class);
        }
      else
        {
  	elf_phdr_set_filesz(&phdr_cp, p_filesz + payload_size + (p_modified_padding - p_og_padding), class);
  	elf_phdr_set_memsz(&phdr_cp, p_memsz + payload_size + (p_modified_padding - p_og_padding), class);
  	// fill between code with \0 or with nop?
        }

      uint64_t og_p_vaddr = elf_get_program_header_vaddr(&phdr, class);

      while (ndx_ph < phnum)
      {
          uint64_t p_vaddr = elf_get_program_header_vaddr(&phdr, class);
          uint64_t p_paddr = elf_get_program_header_paddr(&phdr, class);

          if (p_og_padding < p_modified_padding && og_p_vaddr < p_vaddr)
          {
              elf_phdr_set_vaddr(&phdr_cp, p_vaddr + (p_modified_padding - p_og_padding), class);
              elf_phdr_set_offset(&phdr_cp, p_offset + (p_modified_padding - p_og_padding), class);
              if (p_paddr)
                  elf_phdr_set_paddr(&phdr_cp, p_paddr + (p_modified_padding - p_og_padding), class);
          }

          ,*writed += write(packed, phdr_cp.elf32, phdr_struct_size);
          ndx_ph++;

          if (ndx_ph >= phnum)
              break;

          phdr = elf_get_program_header(ehdr, ndx_ph);
          memcpy(phdr_cp.elf32, phdr.elf32, phdr_struct_size);
      }

      free(phdr_cp.elf32);
      return 0;
  }

  static int write_payload_and_alignment(int packed, void *generated_bytecode, uint64_t payload_size,
                                         uint64_t modified_section_padding)
  {
      if (write(packed, generated_bytecode, payload_size) != (ssize_t)payload_size)
          return 1;

      free(generated_bytecode);

      while (modified_section_padding--)
          write(packed, "\0", 1);

      return 0;
  }

  static int write_remaining_section_headers(t_elf_h *ehdr, int packed, uint64_t ndx_sh, uint64_t shnum,
                                             unsigned char class, uint64_t sh_offset, uint64_t old_next_offset,
                                             uint64_t new_next_offset, uint64_t *writed, uint64_t payload_size)
  {
      uint64_t shdr_struct_size = elf_get_section_header_struct_size(ehdr);
      t_elf_sh shdr_cp = (t_elf_sh){.elf32 = malloc(shdr_struct_size)};
      if (!shdr_cp.elf32)
          return 12;

      t_elf_sh shdr = elf_get_section_header(ehdr, ndx_sh);
      memcpy(shdr_cp.elf32, shdr.elf32, shdr_struct_size);
      elf_shdr_set_size(&shdr_cp, elf_get_section_header_size(&shdr, class) + payload_size, class);
      ,*writed += write(packed, shdr_cp.elf32, shdr_struct_size);

      while (++ndx_sh < shnum)
      {
          shdr = elf_get_section_header(ehdr, ndx_sh);
          uint64_t shcp_addr = elf_get_section_header_addr(&shdr, class);
          uint64_t shcp_offset = elf_get_section_header_offset(&shdr, class);

          memcpy(shdr_cp.elf32, shdr.elf32, shdr_struct_size);

          if (shcp_offset >= (sh_offset + old_next_offset))
          {
              if (shcp_addr > elf_get_section_header_addr(&shdr, class))
                  elf_shdr_set_addr(&shdr_cp, shcp_addr + (new_next_offset - old_next_offset), class);
              if (shcp_offset > sh_offset)
                  elf_shdr_set_offset(&shdr_cp, shcp_offset + (new_next_offset - old_next_offset), class);
          }

          ,*writed += write(packed, shdr_cp.elf32, shdr_struct_size);
      }

      free(shdr_cp.elf32);
      return 0;
  }

  int generated_packed_file(void *elf_mmap, size_t file_size, void *generated_bytecode, size_t payload_size, int packed)
  {
      t_elf_h ehdr = (t_elf_h){.elf32 = elf_mmap};
      uint64_t shnum;
      uint64_t phnum;
      uint64_t ndx_sh;
      uint64_t ndx_ph;
      unsigned char class;
      uint64_t e_entrypoint = elf_get_elfh_entry(&ehdr);
      uint64_t e_shoff = elf_get_elfh_shoff(&ehdr);
      uint64_t sh_offset_payload;
      uint64_t sh_elf_offset;
      uint64_t sh_addralign;
      uint64_t sh_size;
      uint64_t sh_offset;
      uint64_t sh_addr;
      uint64_t old_next_offset;
      uint64_t new_next_offset;
      uint64_t writed = 0;

      if (validate_elf_headers(&ehdr, &shnum, &phnum, &class))
          return 1;
      if (find_section_and_program_indices(&ehdr, e_entrypoint, shnum, phnum, class, &ndx_sh, &ndx_ph))
          return 10;
      if (prepare_info(&ehdr, class, payload_size,
  		     &old_next_offset, phnum, &new_next_offset))
          return 11;
      if (copy_and_modify_elf_header(&ehdr, packed, e_shoff, e_entrypoint, sh_size, class,
                                     old_next_offset, new_next_offset, &writed))
          return 14;
      if (write_program_headers(&ehdr, elf_mmap, packed, ndx_ph, phnum, class, payload_size, &writed, ndx_sh))
          return 15;
      writed += write(packed, elf_mmap + writed, sh_offset_payload - writed); //write data until payload offset.
      if (write_payload_and_alignment(packed, generated_bytecode, payload_size,
                                      new_next_offset - (sh_size + payload_size)))
          return 1;
      writed += old_next_offset - sh_size;
      writed += write(packed, elf_mmap + writed, sh_elf_offset - writed);
      printf("Successfully packed\n");
      if (write_remaining_section_headers(&ehdr, packed, ndx_sh, shnum, class,
                                          sh_offset, old_next_offset, new_next_offset, &writed, payload_size))
          return 12;
      writed += write(packed, elf_mmap + writed, file_size - writed);//Write the rest of the file
      printf("Successfully packed\n");
      return 0;
  }
