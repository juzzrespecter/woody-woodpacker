  {
  	t_list	*node;

  void	ft_lstdelone(t_list *lst, void (*del)(void *))
  {
  	if (!lst || !del)
  		return ;
  	del(lst->content);
  	free(lst);
  }

  void	ft_lstclear(t_list **lst, void (*del)(void *))
  {
  	t_list	*node;
  	t_list	*tmp;

  	if (!lst || !*lst)
  		return ;
  	node = *lst;
  	while (node)
  	{
  		tmp = node->next;
  		ft_lstdelone(node, del);
  		node = tmp;
  	}
  	,*lst = NULL;
  }

  t_list	*ft_lstlast(t_list *lst)
  {
  	if (!lst)
  		return (NULL);
  	while (lst->next)
  		lst = lst->next;
  	return (lst);
  }

  void	ft_lstadd_back(t_list **lst, t_list *new)
  {
  	t_list	*lst_last;

  	if (!new || !lst)
  		return ;
  	if (*lst == NULL)
  		,*lst = new;
  	else
  	{
  		lst_last = ft_lstlast(*lst);
  		lst_last->next = new;
  	}
  }

        t_lz77 *token = (t_lz77 *)tmp->content;
          
        uint16_t dist_len = token->distance_len;
        str_pos[0] = dist_len & 0xFF;         // low byte

  /**
   ,* @brief Compresses data using LZ77 algorithm and returns a list of t_lz77 tokens.
   ,*
   ,* @param data Input data to compress
   ,* @param data_len Length of input data
   ,* @return t_list* Head of the list containing t_lz77 tokens, or NULL on error
   ,*/
  t_list *lz77_compress_to_list(const unsigned char *data, size_t data_len) {
      if (!data || data_len == 0) return NULL;

      t_list *lst = NULL;
      size_t i = 0;

      while (i < data_len) {
          size_t best_offset = 0;
          size_t best_length = 0;

          // Define window start
          size_t window_start = (i > COMPRESS_LZ77_SLIDING_WINDOW_SIZE) ?
                                i - COMPRESS_LZ77_SLIDING_WINDOW_SIZE : 0;

          // Search for the longest match in sliding window
          for (size_t offset = 1; offset <= (i - window_start); offset++) {
              size_t match_length = 0;
              size_t max_possible = (data_len - i > COMPRESS_LZ77_MAX_MATCH_LEN) ?
                                    COMPRESS_LZ77_MAX_MATCH_LEN : (data_len - i);

              while (match_length < max_possible &&
                     data[i - offset + match_length] == data[i + match_length]) {
                  match_length++;
              }

              if (match_length >= COMPRESS_LZ77_MIN_MATCH_LEN && match_length > best_length) {
                  best_length = match_length;
                  best_offset = offset;
              }
          }

          // Allocate token
          t_lz77 *token = __builtin_malloc(sizeof(t_lz77));
          if (!token) {
              ft_lstclear(&lst, free);
              return NULL;
          }
          token->distance_len = 0;
          token->byte = 0;

          if (best_length >= COMPRESS_LZ77_MIN_MATCH_LEN) {
              // Valid match
              if (best_offset > 0xFFF) {
                  fprintf(stderr, "Offset %zu exceeds 12-bit limit\n", best_offset);
                  free(token);
                  ft_lstclear(&lst, free);
                  return NULL;
              }
              size_t token_len = best_length - 2; // becomes 1..15
              token->distance_len = (uint16_t)((best_offset << 4) | (token_len & 0x0F));
              token->byte = (i + best_length < data_len) ? data[i + best_length] : 0;
              i += best_length + 1;
          } else {
              // Literal byte
              token->distance_len = 0; // indicates literal
              token->byte = data[i];
              i += 1;
          }

          // Create list node
          t_list *node = ft_lstnew(token);
          if (!node) {
              free(token);
              ft_lstclear(&lst, free);
              return NULL;
          }

          ft_lstadd_back(&lst, node);

          // Optional debug print
          /*
          printf("Token: distlen=0x%04x (offset=%zu, len=%zu), byte=0x%02x at i=%zu\n",
                 token->distance_len,
                 (size_t)(token->distance_len >> 4),
                 (size_t)((token->distance_len & 0xF) ? ((token->distance_len & 0xF) + 2) : 0),
                 token->byte,
                 i - (best_length >= 3 ? best_length + 1 : 1));
          ,*/
      }

      return lst;
  }


  int lz77_decompress(const uint8_t* in, size_t in_size, uint8_t* out)
  {
      if (in_size <= 7 || out == NULL || in == NULL) {
          return -1;
      }

      uint64_t orig_size = *(const uint64_t*)in;
      if (orig_size == 0) {
          return -1;
      }

      const uint8_t* p = in + 11;  // Start at in+8, then advance 3 for first iteration
      const uint8_t* in_end = in + in_size;
      size_t pos = 0;

      while (p <= in_end && pos < orig_size) {
          
          // Read token from [p-3, p-2] 
          uint16_t token = *(const uint16_t*)(p - 3);
          uint8_t length = token & 0xF;
          uint16_t distance = token >> 4;
          
          if (length || distance || pos >= distance)
  	  {
              // Copy match
              length += 2; // Adjust length
              
              // Ensure bounds
              if (pos + length > orig_size) {
  	      length = orig_size - pos;
              }
              
              // Copy match data (handle overlapping case)
              size_t src = pos - distance;
              for (uint16_t i = 0; i < length; i++) {
  	      out[pos + i] = out[src + i];
              }
              pos += length;
  	  }
  	// Add literal if space allows
  	if (pos < orig_size) {
  	  out[pos++] = *(p - 1);
  	}
  	p += 3;
      }
      return (pos == orig_size) ? 0 : -1;
  }

  size_t set_xor_register_register(void *dst, t_registers reg, unsigned char class)
  {
    const char bytecode_regmap[] = {'\xc0', '\xc0', '\xdb', '\xdb', '\xc9', '\xc9', '\xd2', '\xd2', '\xff', '\xff', '\xf6', '\xf6', '\xed', '\xed', '\xe4', '\xe4', '\xc0', '\xc9', '\xd2','\xdb','\xe4', '\xed', '\xf6', '\xff'};
    size_t bytes;

    if (reg > R15 || (ELFCLASS32 == class && reg >= R8))
      return (-1);
    bytes = 0;
    if (ELFCLASS64 == class)
      {
        if (dst && reg >= R8 && reg <= R15)
  	memcpy(dst + bytes, "\x4D", 1);
        else if (dst)
  	memcpy(dst + bytes, "\x48", 1);
        bytes++;
      }
    if (dst)
      memcpy(dst + bytes, "\x31", 1);
    bytes++;
    if (dst)
      memcpy(dst + bytes, &bytecode_regmap[reg], 1);
    return (++bytes);
  }

  size_t set_xor_register_size(void *dst, unsigned char class)
  {
    size_t xor_register_size;
    t_registers reg;

    if (ELFCLASS64 == class)
      reg = RAX;// RAX gonna hold the value of entrypoint.
    else
      reg = EAX;
    xor_register_size = 0;
    while ((ELFCLASS64 == class && reg <= R15) || reg < ESP)
      {
        if (reg == R13)
  	reg++;
        if (dst)
  	xor_register_size += set_xor_register_register(dst + xor_register_size, reg, class);
        else
  	xor_register_size += set_xor_register_register(NULL, reg, class);
        if (reg >= R8)
  	reg++; 
        else
  	reg += 2;
        if (reg == RSP)//No xor rsp
  	reg += 2;
      }
    return (xor_register_size);
  }

  size_t set_jmp_entrypoint(void *payload, uint64_t entrypoint, unsigned char class)
  {
    t_elf_addr u_entrypoint = (t_elf_addr){.b64 = entrypoint};
    size_t offset = 0;

    if (class == ELFCLASS64)
      {
        /* memcpy(payload, "\x48", 1); //rex prefix x64 */
        if (payload)
  	memcpy(payload, "\x49", 1);
        offset++;
        if (payload)
  	memcpy(payload + offset, "\xbd", 1); //mov r13
        offset++;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b64, sizeof (u_entrypoint.b64));
        offset += sizeof (u_entrypoint.b64);
        if (payload)
  	{
  	offset += pop_r13(payload + offset);
  	offset += add_r13(payload + offset, u_entrypoint.b64);
  	offset += set_jmp_r13(payload + offset);
  	}
        else
  	{
  	offset += pop_r13(NULL);
  	offset += add_r13(NULL, u_entrypoint.b64);
  	offset += set_jmp_r13(NULL);
  	}
      }
    else
      {
        if (payload)
  	memcpy(payload + offset, "\x81\x04\x24", 3);// add DWORD [esp]
        offset+= 3;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b32, sizeof (u_entrypoint.b32));
        offset += sizeof (u_entrypoint.b32);
        if (payload)
  	memcpy(payload + offset, "\xc3", 1);//ret
        offset++;
      }
    return (offset);
  }

  {

    uint64_t new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    uint64_t new_phdr_offset = elf_get_highest_offset_avalaible(ehdr, class, shnum);

    const uint64_t new_align = 0x1000;
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    if (increment_offset_phdr(ehdr, new_phdr_offset, phnum, class))// I think shouldn't go throught this way.
      new_phdr_offset += phdr_struct_size;
    if (new_phdr_vaddr % new_align == new_phdr_offset % new_align) 
      printf("all fine with addr and offset aligned\n");
    else
      {
        if (new_phdr_vaddr % new_align > new_phdr_offset % new_align)
  	new_phdr_offset += new_phdr_vaddr % new_align - new_phdr_offset % new_align;
        else
  	new_phdr_vaddr += new_phdr_offset % new_align - new_phdr_vaddr % new_align;

      {
        new_phdr_vaddr += 16 - new_phdr_vaddr % 16;
        new_phdr_offset += 16 - new_phdr_offset % 16;
      }
    return ((uint32_t)new_phdr_vaddr);
  }

  int load_stage_buffer(int class, char **stage_buffer, size_t *stage_fd_size, char *filename_x86, char *filename_x64)
  {
      int stage_fd;
      
      // Validate input parameters
      if (!stage_buffer || !stage_fd_size)
          return (1);
      
      // Initialize output parameters
      ,*stage_buffer = NULL;
      ,*stage_fd_size = 0;
      
      stage_fd = open(ft_ternary(ELFCLASS32 == class, filename_x86,filename_x64), O_RDONLY);
      if (stage_fd < 0 || get_file_size(stage_fd, (off_t *)stage_fd_size) || !*stage_fd_size)
      {
          if (stage_fd >= 0)
              close(stage_fd);
          return (1);
      }
      
      ,*stage_buffer = malloc(*stage_fd_size);
      if (!(*stage_buffer))
      {
          close(stage_fd);
          return (5);
      }

      {
          free(*stage_buffer);
          ,*stage_buffer = NULL;
          close(stage_fd);
          return (5);
      }
      
      close(stage_fd);
      return (0);
  }

  static uint64_t elf_get_closest_higher_p_offset(const t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t p_offset)

  }

  	      continue;
  	}
        cur_vaddr_top = elf_get_program_header_vaddr(&phdr, class) + elf_get_program_header_memsz(&phdr, class);
        top_vaddr = get_alignment_effective_addr(0x1000, cur_vaddr_top);
        avalaible_vaddr = top_vaddr - cur_vaddr_top;
        if (avalaible_vaddr < gap_size)
  	{
  	  ph_ndx++;
  	  continue;
  	}
        cur_offset_top = elf_get_program_header_offset(&phdr, class) + elf_get_program_header_filesz(&phdr, class);
        top_offset = elf_get_closest_higher_p_offset(ehdr, class, phnum, cur_offset_top);
        if (top_offset == (uint64_t) -1)
  	top_offset = elf_get_elfh_shoff(ehdr);
        avalaible_offset = top_offset - cur_offset_top;
        if (avalaible_offset >= gap_size)
  	{
  	  ,*ret_offset = avalaible_offset;
  	  ,*ret_vaddr = cur_vaddr_top - 1;
  	  return (true);
  	}
        ph_ndx++;
      }
    return (false);
  }

  int generate_payload(void *elf_mmap, size_t elf_size, char **payload_ptr, size_t *payload_size, char **stage3, size_t *stage3_size, void **new_elf_buffer)

    const t_elf_h ehdr = (const t_elf_h){.elf32 = elf_mmap};

    off_t stage3_fd_size;
    char *stage3_buffer;

    uint64_t avalaible_vaddr = 0;
    uint64_t avalaible_offset = 0;

    uint64_t ehdr_struct_size;

    if (load_stage_buffer(class, &stage3_buffer, (size_t *)&stage3_fd_size, "bytecode/x86/stage3.bin", "bytecode/x64/stage3.bin"))
        return (5);
    if (elf_is_a_avalaible_gap_in_exec_load_phdr((void *)&ehdr, class, phnum, stage3_fd_size, &avalaible_vaddr, &avalaible_offset))
      {
        ,*new_elf_buffer = malloc(elf_size);
        if (!*new_elf_buffer)
  	{
  	  free(stage3_buffer);
  	  return (5);
  	}
        memcpy(*new_elf_buffer, elf_mmap, elf_size);
        memcpy(*new_elf_buffer + avalaible_offset, stage3_buffer, stage3_fd_size);
        free(stage3_buffer);
        ndx_ph = elf_get_program_ndx_from_addr((void *)&ehdr, class, phnum, avalaible_vaddr);
        if (ndx_ph >= phnum)
  	  return (5);
        ehdr_struct_size = elf_get_struct_size_elf_header(elf_mmap);
        phdr =(t_elf_ph){.elf32 = *new_elf_buffer + (elf_get_struct_size_elf_ph(class) * ndx_ph) + ehdr_struct_size};

      {

        return (5);

      }

    size_t stage1_stage2_elf_size = stage2_elf_compress_encrypt_header_size + stage1_fd_size;
    char *stage1_stage2_elf = malloc(stage1_stage2_elf_size);
    if (!stage1_stage2_elf)

        free(stage1_buffer);
        free(stage2_elf_compress_encrypt_header);
        return (5);
      }
    memcpy(stage1_stage2_elf, stage1_buffer, stage1_fd_size);
    free(stage1_buffer);
    memcpy(stage1_stage2_elf + stage1_fd_size, stage2_elf_compress_encrypt_header, stage2_elf_compress_encrypt_header_size);
    free(stage2_elf_compress_encrypt_header);
   ,*payload_size = stage1_stage2_elf_size;
    ,*payload_ptr = stage1_stage2_elf;
    return (0);
  }

  bool padding(int packed, size_t num)
  {
    bool i = true;
    size_t times = 0;

    while (i && times < num)
      {
        if ((ssize_t)1 != write(packed, "\0", 1))
  	i = false;
        times++;
      }
    return (!i);
  }

  static int write_program_headers(t_elf_h *ehdr, int packed, uint64_t phnum, uint64_t new_phdr_vaddr,
                                   unsigned char class, uint64_t payload_size, uint64_t *writed/* , uint64_t *stage3_size */)
  {
    const uint64_t ehdr_struct_size = elf_get_struct_size_elf_header((char *)ehdr->elf32);
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    t_elf_ph phdr;
    uint64_t ndx_ph;
    uint64_t p_offset;
    uint64_t p_filesz;
    uint64_t p_memsz;
    uint64_t p_align;
    uint64_t p_vaddr;
    uint64_t p_paddr;
    uint64_t p_type;
    uint64_t p_flags;
    t_elf_ph phdr_cp = {.elf32 = malloc(phdr_struct_size)};

    if (!phdr_cp.elf32)
      return (1);
    ndx_ph = 0;
    phdr = elf_get_program_header(ehdr, ndx_ph);

    p_align = 0x1000;
    p_filesz = p_align;
    p_memsz = elf_get_highest_used_vaddr_avalaible(ehdr, class, phnum);
    p_offset = 0;
    p_vaddr = elf_get_program_header_vaddr(&phdr, class) & ~(p_align - 1);
    p_memsz -= p_vaddr;
    p_paddr = elf_get_program_header_paddr(&phdr, class) & ~(p_align - 1);
    p_type = PT_LOAD;
    p_flags = PF_R | PF_W;
    elf_phdr_set_filesz(&phdr_cp, p_filesz, class);
    elf_phdr_set_memsz(&phdr_cp, p_memsz, class);
    elf_phdr_set_offset(&phdr_cp, p_offset, class);
    elf_phdr_set_vaddr(&phdr_cp, p_vaddr, class);
    elf_phdr_set_paddr(&phdr_cp, p_paddr, class);
    elf_phdr_set_type(&phdr_cp, p_type, class);
    elf_phdr_set_flags(&phdr_cp, p_flags, class);
    elf_phdr_set_align(&phdr_cp, p_align, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      {
        free(phdr_cp.elf32);
        return (2);
      }
    ,*writed += phdr_struct_size;//first header
    p_flags = PF_R | PF_X;
    p_vaddr = new_phdr_vaddr;
    p_paddr = p_vaddr;
    p_memsz = payload_size + ehdr_struct_size + phdr_struct_size * (2 + elf_is_gnu_stack_phdr(ehdr, class, phnum));

    p_filesz = p_memsz;
    elf_phdr_set_filesz(&phdr_cp, p_filesz, class);
    elf_phdr_set_memsz(&phdr_cp, p_memsz, class);
    elf_phdr_set_vaddr(&phdr_cp, p_vaddr, class);
    elf_phdr_set_paddr(&phdr_cp, p_paddr, class);
    elf_phdr_set_flags(&phdr_cp, p_flags, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      {
        free(phdr_cp.elf32);
        return (3);
      }
    ,*writed += phdr_struct_size;//second header
    free(phdr_cp.elf32);
    if (elf_is_gnu_stack_phdr(ehdr, class, phnum))
      {
        ndx_ph = elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum);
        phdr = elf_get_program_header(ehdr, ndx_ph);
        if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
  	  return (4);
        ,*writed += phdr_struct_size;
      }
    return (0);
  }

  static int write_padding_and_payload(int packed, void **generated_bytecode, uint64_t payload_size,
  				     uint64_t padding, uint64_t *writed)
  {
    /* printf("padding: %lu\n", padding); */
    while (padding--)
      {
        if (write(packed, "\0", 1) != 1)
  	return (1);
        writed[0]++;
      }
    if (write(packed, *generated_bytecode, payload_size) != (ssize_t)payload_size)
      {
        free(*generated_bytecode);
        ,*generated_bytecode = NULL;
        return (2);
      }
    free(*generated_bytecode);
    ,*generated_bytecode = NULL;
    ,*writed += payload_size;
    return (0);
  }

      mov    edx, ecx           ; edx = orig_size

      movzx  esi, si            ; Zero-extend distance to esi

      movzx  r11d, edx          ; Zero-extend length to r11d

      xor    esi, esi           ; esi = 0 (loop counter)

      cmp    rax, rcx           ; Compare pos with orig_size
      jb     .process_token     ; Jump back to process next token if pos < orig_size

      pop    r14                ; Restore r14 register
      setne  al                 ; Set AL to 1 if pos != orig_size
      movzx  eax, al            ; Zero-extend AL to EAX

  ; void calculate_elf_memory_range(Elf64_Phdr *phdr, unsigned int phnum, uint64_t *min_vaddr, uint64_t *max_vaddr)
  calculate_elf_memory_range:
      xor    eax, eax                        ; rax = 0
      mov    QWORD  [rdx], 0xffffffffffffffff ; *min_vaddr = UINT64_MAX
      mov    QWORD  [rcx], rax            ; *max_vaddr = 0
      movsxd rax, esi                         ; rax = phnum (sign extend to 64-bit); May optimize with lea?
      imul   rax, rax, 0x38                  ; rax = phnum * sizeof(Elf64_Phdr)
      add    rdi, rax                        ; rdi = phdr + (phnum * sizeof(Elf64_Phdr))
                                              ; rdi now points to the end of the phdr array

  .start_loop:
      test   esi, esi                        ; if (phnum == 0)
      je     .done                           ;   goto done
      dec    esi                             ; phnum--
      cmp    DWORD  [rdi-0x38], 0x1       ; if (phdr[-1].p_type != PT_LOAD)
      jne    .next_iteration                 ;   goto next_iteration

      ; Process PT_LOAD segment
      mov    rax, QWORD  [rdi-0x28]      ; rax = phdr[-1].p_vaddr
      cmp    rax, QWORD  [rdx]            ; if (p_vaddr < *min_vaddr)
      jae    .check_max                      ;   goto check_max
      mov    QWORD  [rdx], rax            ; *min_vaddr = p_vaddr

  .check_max:
      mov    rax, QWORD  [rdi-0x10]      ; rax = phdr[-1].p_memsz
      add    rax, QWORD  [rdi-0x28]       ; rax = p_vaddr + p_memsz
      cmp    QWORD  [rcx], rax            ; if (*max_vaddr < (p_vaddr + p_memsz))
      jae    .next_iteration                 ;   goto next_iteration
      mov    QWORD  [rcx], rax            ; *max_vaddr = p_vaddr + p_memsz

  .next_iteration:
      sub    rdi, 0x38                       ; rdi = phdr - sizeof(Elf64_Phdr) (move to previous phdr)
      jmp    .start_loop                     ; goto start_loop

  .done:
      and    QWORD  [rdx], 0xfffffffffffff000 ; *min_vaddr &= ~(PAGE_SIZE - 1) (align down to page)
      ret                                     ; return

  load_elf_segment:

      cmp    eax, 0x1                 ; Compare with 1

    sub    r13, r15                 ; r13 = (p_offset + p_memsz) - aligned p_offset
    mov    QWORD PTR [rsp], rsi     ; Save elf_ptr on stack
    mov    rbx, rdx                 ; rbx = phdr
    sub    rax, r9                  ; rax = aligned p_vaddr - min_vaddr
    mov    edx, ebp                 ; edx = combined flags/protection
    mov    r9, r15                  ; r9 = aligned p_offset
    mov    rsi, r13                 ; rsi = size
    lea    r14, [rdi+rax*1]         ; r14 = base + (aligned p_vaddr - min_vaddr)
    mov    DWORD PTR [rsp+0xc], r8d ; Save fd on stack
    mov    rdi, r14                 ; rdi = mapping address
    call   mmap                     ; Call mmap syscall
    mov    rdx, rax                 ; rdx = result of mmap
    mov    eax, 0xffffffff          ; Set eax to -1
    cmp    rdx, 0xffffffffffffffff  ; Check if mmap failed
    je     .mmap_failed             ; Jump if mmap failed
    
    ; Check fd value
    cmp    DWORD PTR [rsp+0xc], 0xffffffff ; Check if fd == -1
    jne    .fd_not_negative_one     ; Jump if fd != -1
    
    ; fd == -1: Copy data from ELF buffer to mapped memory
    mov    rsi, QWORD PTR [rsp]     ; rsi = elf_ptr
    mov    rcx, QWORD PTR [rbx+0x20] ; rcx = p_memsz
    ret                            ; Return
