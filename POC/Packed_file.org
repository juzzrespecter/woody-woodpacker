** Scheme view of the struct of our packed file
| ehdr | phdr | s1+crypt_compress_s2+elf_file | end |
- ehdr -> modified {discard all shs data, modify phnum, entry}.
- phdr -> one phdr that use all memsz of og bin, a new one for s1(stage 1), maybe a gnu_stack.
- s1 + crypt_compress_s2 + elf_file
  - s1: get aslr, va, size s1, get fd of /proc/self/exe, mmap s2 RW-, decrypt, decompress in s2, mprotect s2, jmp s2.
  - s2: mmap with fd, close fd, munmap s1, load elf, load interp, set auxv, jmp s3, payload.
  - s3: munmap s2, reset r/edx, jmp to entrypoint(interpe or elf).  if couldn't append to one phdr R-X of the elf file s2 create one.

  unsigned char *lz77_get_compress_arr(t_list *lst, size_t *arr_size)
  {
      ,*arr_size = ft_lstsize(lst) * 3;
      unsigned char *str = malloc(*arr_size);
      unsigned char *str_pos = str;
      if (!str)
  	return (str);

      t_list *tmp = lst;
      while (tmp)
  	{
  	    t_lz77 *token = (t_lz77 *)tmp->content;

  	    uint16_t dist_len = token->distance_len;
  	    str_pos[0] = dist_len & 0xFF;         // low byte
  	    str_pos[1] = (dist_len >> 8) & 0xFF;  // high byte  
  	    str_pos[2] = token->byte;             // literal byte
  	    str_pos += 3;
  	    tmp = tmp->next;
  	}
      return ((void *)str);
  }

  /**
   ,* @brief Compresses data using LZ77 algorithm and returns a list of t_lz77 tokens.
   ,*
   ,* @param data Input data to compress
   ,* @param data_len Length of input data
   ,* @return t_list* Head of the list containing t_lz77 tokens, or NULL on error
   ,*/

  t_list *lz77_compress_to_list(const unsigned char *data, size_t data_len) {
      if (!data || data_len == 0) return NULL;

      t_list *lst = NULL;
      size_t i = 0;

      while (i < data_len) {
  	size_t best_offset = 0;
  	size_t best_length = 0;

  	// Define window start
  	size_t window_start = (i > COMPRESS_LZ77_SLIDING_WINDOW_SIZE) ?
  	    i - COMPRESS_LZ77_SLIDING_WINDOW_SIZE : 0;

  	// Search for the longest match in sliding window
  	for (size_t offset = 1; offset <= (i - window_start) && offset <= COMPRESS_LZ77_OFFSET_MAX_LIMIT; offset++) {
  	    size_t match_length = 0;
  	    size_t max_possible = (data_len - i > COMPRESS_LZ77_MAX_MATCH_LEN) ?
  		COMPRESS_LZ77_MAX_MATCH_LEN : (data_len - i);

  	    while (match_length < max_possible && 
  		   data[i - offset + match_length] == data[i + match_length]) {
  		match_length++;
  	    }

  	    if (match_length >= COMPRESS_LZ77_MIN_MATCH_LEN && match_length > best_length) {
  		best_length = match_length;
  		best_offset = offset;
  	    }
  	}

  	// Allocate token
  	t_lz77 *token = malloc(sizeof(t_lz77));
  	if (!token) {
  	    ft_lstclear(&lst, free);
  	    return NULL;
  	}
  	token->distance_len = 0;
  	token->byte = 0;

  	if (best_length >= COMPRESS_LZ77_MIN_MATCH_LEN) {
  	    // Valid match
  	    if (best_offset > 0xFFF) {
  		/* fprintf(stderr, "Offset %zu exceeds 12-bit limit\n", best_offset); */
  		free(token);
  		ft_lstclear(&lst, free);
  		return NULL;
  	    }
  	    size_t token_len = best_length - 2;
  	    token->distance_len = (uint16_t)((best_offset << 4) | (token_len & 0x0F));
  	    token->byte = (i + best_length < data_len) ? data[i + best_length] : 0;
  	    i += best_length + 1;
  	} else {
  	    token->distance_len = 0;
  	    token->byte = data[i];
  	    i += 1;
  	}

  	// Create list node
  	t_list *node = ft_lstnew(token);
  	if (!node) {
  	    free(token);
  	    ft_lstclear(&lst, free);
  	    return NULL;
  	}

  	ft_lstadd_back(&lst, node);
      }

      return lst;
  }


  int lz77_decompress(const uint8_t* in, size_t in_size, uint8_t* out)
  {
      if (in_size <= 7 || out == NULL || in == NULL) {
  	return -1;
      }

      uint64_t orig_size = *(const uint64_t*)in;
      if (orig_size == 0) {
  	return -1;
      }

      const uint8_t* p = in + 11;  // Start at in+8, then advance 3 for first iteration
      const uint8_t* in_end = in + in_size;
      size_t pos = 0;

      while (p <= in_end && pos < orig_size) {

  	// Read token from [p-3, p-2] 
  	uint16_t token = *(const uint16_t*)(p - 3);
  	uint8_t length = token & 0xF;
  	uint16_t distance = token >> 4;

  	if (length || distance || pos >= distance)
  	    {
  		// Copy match
  		length += 2; // Adjust length

  		// Ensure bounds
  		if (pos + length > orig_size) {
  		    length = orig_size - pos;
  		}

  		// Copy match data (handle overlapping case)
  		size_t src = pos - distance;
  		for (uint16_t i = 0; i < length; i++) {
  		    out[pos + i] = out[src + i];
  		}
  		pos += length;
  	    }
  	// Add literal if space allows
  	if (pos < orig_size) {
  	    out[pos++] = *(p - 1);
  	}
  	p += 3;
      }
      return (pos == orig_size) ? 0 : -1;
  }

  size_t ft_je(unsigned char *dst, size_t distance)
  {
      t_type hop = (t_type){.b8 = distance};
      size_t written;

      written = 0;

  {
      size_t	written;
      const t_type	type = (t_type){.b8 = value};

      if (ELFCLASS32 == class && type.b8 != type.b4)
  	return (-1);
      written = 0;
      if (class == ELFCLASS64)
  	{
  	    if (dst)
  		ft_memcpy(dst, "\x48", 1);
  	    written++;
  	}
      if ((uint32_t)type.b  == type.b4 && (uint64_t)type.b4 == type.b8 && type.b < 0x80)
  	{
  	    if (dst)
  		ft_memcpy(dst + written, "\x83", 1);
  	    written++;
  	    if (dst)
  		{
  		    /* if (ELFCLASS64 == class) */
  		    ft_memcpy(dst+written, "\xec", 1);
  		    /* else */
  		    /* ft_memcpy(dst+written, "\xe8", 1); */
  		}
  	    written++;
  	    if (dst)
  		ft_memcpy(dst + written, &type.b, sizeof (type.b));
  	    written += sizeof(type.b);
  	}
      else
  	{
  	    if (dst)
  		{
  		    if (ELFCLASS64 == class)
  			{
  			    ft_memcpy(dst + written, "\x81\xec", 2);
  			    written ++;
  			}
  		    else
  			ft_memcpy(dst + written, "\x2d", 1);
  		}
  	    written ++;
  	    if ((uint64_t)type.b4 == type.b8 && type.b4 < 0x80000000)
  		{
  		    if (dst)
  			ft_memcpy(dst + written, &type.b4, sizeof (type.b4));

  		}
  	    else
  		{
  		    if (dst)
  			ft_memcpy(dst + written, &type.b8, sizeof (type.b8));
  		    written += sizeof(type.b8);
  		}
  	}
      return (written);
  }

  size_t set_xor_register_register(void *dst, t_registers reg, unsigned char class)
  {
      const char bytecode_regmap[] = {'\xc0', '\xc0', '\xdb', '\xdb', '\xc9', '\xc9', '\xd2', '\xd2', '\xff', '\xff', '\xf6', '\xf6', '\xed', '\xed', '\xe4', '\xe4', '\xc0', '\xc9', '\xd2','\xdb','\xe4', '\xed', '\xf6', '\xff'};
      size_t bytes;

      if (reg > R15 || (ELFCLASS32 == class && reg >= R8))
  	return (-1);
      bytes = 0;
      if (ELFCLASS64 == class)
  	{
  	    if (dst && reg >= R8 && reg <= R15)
  		ft_memcpy(dst + bytes, "\x4D", 1);
  	    else if (dst)
  		ft_memcpy(dst + bytes, "\x48", 1);
  	    bytes++;
  	}
      if (dst)
  	ft_memcpy(dst + bytes, "\x31", 1);
      bytes++;
      if (dst)
  	ft_memcpy(dst + bytes, &bytecode_regmap[reg], 1);
      return (++bytes);
  }

  size_t set_xor_register_size(void *dst, unsigned char class)
  {
      size_t xor_register_size;
      t_registers reg;

      if (ELFCLASS64 == class)
  	reg = RAX;// RAX gonna hold the value of entrypoint.
      else
  	reg = EAX;
      xor_register_size = 0;
      while ((ELFCLASS64 == class && reg <= R15) || reg < ESP)
  	{
  	    if (reg == R13)
  		reg++;
  	    if (dst)
  		xor_register_size += set_xor_register_register(dst + xor_register_size, reg, class);
  	    else
  		xor_register_size += set_xor_register_register(NULL, reg, class);
  	    if (reg >= R8)
  		reg++; 
  	    else
  		reg += 2;
  	    if (reg == RSP)//No xor rsp
  		reg += 2;
  	}
      return (xor_register_size);
  }

  size_t set_jmp_entrypoint(void *payload, uint64_t entrypoint, unsigned char class)
  {
      t_elf_addr u_entrypoint = (t_elf_addr){.b64 = entrypoint};
      size_t offset = 0;

      if (class == ELFCLASS64)
  	{
  	    /* ft_memcpy(payload, "\x48", 1); //rex prefix x64 */
  	    if (payload)
  		ft_memcpy(payload, "\x49", 1);
  	    offset++;
  	    if (payload)
  		ft_memcpy(payload + offset, "\xbd", 1); //mov r13
  	    offset++;
  	    if (payload)
  		ft_memcpy(payload + offset, &u_entrypoint.b64, sizeof (u_entrypoint.b64));
  	    offset += sizeof (u_entrypoint.b64);
  	    if (payload)
  		{
  		    offset += pop_r13(payload + offset);
  		    offset += add_r13(payload + offset, u_entrypoint.b64);
  		    offset += set_jmp_r13(payload + offset);
  		}
  	    else
  		{
  		    offset += pop_r13(NULL);
  		    offset += add_r13(NULL, u_entrypoint.b64);
  		    offset += set_jmp_r13(NULL);
  		}
  	}
      else
  	{
  	    if (payload)
  		ft_memcpy(payload + offset, "\x81\x04\x24", 3);// add DWORD [esp]
  	    offset+= 3;
  	    if (payload)
  		ft_memcpy(payload + offset, &u_entrypoint.b32, sizeof (u_entrypoint.b32));
  	    offset += sizeof (u_entrypoint.b32);
  	    if (payload)
  		ft_memcpy(payload + offset, "\xc3", 1);//ret
  	    offset++;
  	}
      return (offset);
  }

  static t_elf_ph get_n_loadable_and_exec_phdr(t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t n)
  {
      t_elf_ph phdr;
      uint64_t ph_ndx;
      uint64_t times_loadable_and_exec;

      ph_ndx = 0;
      times_loadable_and_exec = 0;
      while (++ph_ndx < phnum)
  	{
  	    phdr = elf_get_program_header(ehdr, ph_ndx);
  	    if (PT_LOAD == elf_get_program_header_type(&phdr, class) && PF_X & elf_get_program_header_flags(&phdr, class))
  		times_loadable_and_exec++;
  	    if (times_loadable_and_exec == n)
  		return (phdr);
  	}
      phdr.elf32 = NULL;
      return (phdr);
  }

  static bool increment_offset_phdr(t_elf_h *ehdr, uint64_t offset, uint64_t phnum, unsigned char class)
  {

      t_elf_ph phdr_loadable_and_exec_first = get_n_loadable_and_exec_phdr(ehdr, class, phnum, 1);
      uint64_t phdr_loadable_and_exec_first_offset = elf_get_program_header_offset(&phdr_loadable_and_exec_first, class);

      return (offset > ehdr_struct_size && offset < phdr_loadable_and_exec_first_offset);
  }

  uint32_t get_new_e_entrypoint(t_elf_h *ehdr, unsigned char class)
  {
      uint64_t phnum = elf_get_elfh_phnum(ehdr);
      uint64_t shnum = elf_get_elfh_shnum(ehdr);
      uint64_t new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
      uint64_t new_phdr_offset = elf_get_highest_offset_avalaible(ehdr, class, shnum);

      const uint64_t new_align = 0x1000;
      const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
      new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
      if (increment_offset_phdr(ehdr, new_phdr_offset, phnum, class))// I think shouldn't go throught this way.
  	new_phdr_offset += phdr_struct_size;
      if (new_phdr_vaddr % new_align != new_phdr_offset % new_align) 
  	{
  	    if (new_phdr_vaddr % new_align > new_phdr_offset % new_align)
  		new_phdr_offset += new_phdr_vaddr % new_align - new_phdr_offset % new_align;
  	    else
  		new_phdr_vaddr += new_phdr_offset % new_align - new_phdr_vaddr % new_align;

  	{
  	    new_phdr_vaddr += 16 - new_phdr_vaddr % 16;
  	    new_phdr_offset += 16 - new_phdr_offset % 16;
  	}
      return ((uint32_t)new_phdr_vaddr);
  }

  static uint64_t elf_get_closest_higher_p_offset(const t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t p_offset)
  {

      ph_ndx = 0;

  	    phdr = elf_get_program_header(ehdr, ph_ndx);

      uint64_t cur_vaddr_top;//cur
      uint64_t cur_offset_top;//cur
      uint64_t top_vaddr;
      uint64_t top_offset;//get closest higher offset.
      uint64_t avalaible_vaddr;
      uint64_t avalaible_offset;

      ph_ndx = 0;
      while (ph_ndx < phnum)
  	{
  	    ,*phdr = elf_get_program_header(ehdr, ph_ndx);
  	    if (!(PT_LOAD == elf_get_program_header_type(phdr, class) && PF_X & elf_get_program_header_flags(phdr, class)))
  		{
  		    ph_ndx++;
  		    continue;
  		}
  	    cur_vaddr_top = elf_get_program_header_vaddr(phdr, class) + elf_get_program_header_memsz(phdr, class);
  	    top_vaddr = get_alignment_effective_addr(0x1000, cur_vaddr_top);
  	    avalaible_vaddr = top_vaddr - cur_vaddr_top;
  	    if (avalaible_vaddr < gap_size)
  		{
  		    ph_ndx++;
  		    continue;
  		}
  	    cur_offset_top = elf_get_program_header_offset(phdr, class) + elf_get_program_header_memsz(phdr, class);
  	    top_offset = elf_get_closest_higher_p_offset(ehdr, class, phnum, cur_offset_top);
  	    if (top_offset == (uint64_t) -1)
  		top_offset = elf_get_elfh_shoff(ehdr);
  	    avalaible_offset = top_offset - cur_offset_top;
  	    if (avalaible_offset >= gap_size)
  		{
  		    ,*ret_offset = cur_offset_top;
  		    ,*ret_vaddr = cur_vaddr_top;
  		    return (true);
  		}
  	    ph_ndx++;
  	}
      return (false);
  }

  int load_stage_buffer(t_arch arch, char **stage_buffer, size_t *stage_fd_size, t_bin_type bin_type)
  {
      int stage_fd;

      // Validate input parameters
      if (!stage_buffer || !stage_fd_size)
  	return (1);

      // Initialize output parameters
      ,*stage_buffer = NULL;
      ,*stage_fd_size = 0;

      stage_fd = open(g_binary_paths[arch][bin_type], O_RDONLY);
      if (stage_fd < 0 || get_file_size(stage_fd, (off_t *)stage_fd_size) || !*stage_fd_size)
  	{
  	    if (stage_fd >= 0)
  		close(stage_fd);
  	    return (1);
  	}

      ,*stage_buffer = malloc(*stage_fd_size);
      if (!(*stage_buffer))
  	{
  	    close(stage_fd);
  	    return (5);
  	}

      if (read(stage_fd, *stage_buffer, *stage_fd_size) != (ssize_t) *stage_fd_size)
  	{
  	    free(*stage_buffer);
  	    ,*stage_buffer = NULL;
  	    close(stage_fd);
  	    return (5);
  	}

      close(stage_fd);
      return (0);
  }

          ft_memcpy(buffer + offsets[i], &values[i].b8, data_size);
      }
  }

  /**
   ,* @brief Loads a stage binary and patches it with provided values.
   ,*
   ,* @param arch              x86/x64
   ,* @param bin_type	    Bin Type
   ,* @param patch_offsets     Array of offset locations to patch (per architecture)
   ,* @param patch_values      Array of t_type values to write at those offsets
   ,* @param patch_count       Number of patches to apply
   ,* @param loaded_buffer     Loaded_buffer: allocated patched buffer (caller must eventually free)
   ,* @param loaded_size       Loaded_size: size of loaded binary
   ,* @return                  0 on success, non-zero on error
   ,*/
  static int load_and_apply_patches_to_buffer(t_arch arch, char **loaded_buffer, size_t *loaded_size,
  					    t_bin_type bin_type, const size_t *patch_offsets, const t_type *patch_values, size_t patch_count)
  {
      // Step 1: Load raw binary
      if (load_stage_buffer(arch, loaded_buffer, loaded_size, bin_type))
  	{
  	    ,*loaded_buffer = NULL;
  	    return (5); // Load failed
  	}
      // Step 2: Determine data size for patching (32 vs 64-bit pointers/values)
      size_t value_size = (size_t)ft_ternary(arch, (void *)sizeof(uint64_t), (void *)sizeof(uint32_t));
      // Step 3: Apply all patches
      apply_patches_to_buffer(*loaded_buffer, patch_offsets, patch_values, patch_count, value_size);
      return 0; // Success
  }


  void *buffer_join(size_t *join_len, void *s1, size_t len_s1, void *s2, size_t len_s2)
  {
      void *join;

      ,*join_len = len_s2 + len_s1;
      join = malloc(*join_len);
      if (!join)
  	return (join);
      ft_memcpy(join, s1, len_s1);
      ft_memcpy(join+len_s1, s2, len_s2);
      return (join);
  }

  /**
   ,* @brief Compresses input data using LZ77, adds size header, then encrypts.
   ,* @return Encrypted output buffer, or NULL on error.
   ,*/
  static char *compress_encrypt_with_header(const void *input_data, size_t input_size,
  					  size_t value_size, size_t *output_size,
  					  size_t *final_compressed_size, char **key)
  {
      // Step 1: Compress
      t_list *compressed_list = lz77_compress_to_list((unsigned char *)input_data, input_size);
      if (!compressed_list) return NULL;

      size_t compressed_size;
      unsigned char *compressed_bytes = lz77_get_compress_arr(compressed_list, &compressed_size);
      ft_lstclear(&compressed_list, free);
      if (!compressed_bytes) return NULL;

      // Step 2: Prepend original size as header
      char *compressed_with_header = buffer_join(
  					       final_compressed_size,
  					       &input_size, value_size,
  					       compressed_bytes, compressed_size
  					       );
      free(compressed_bytes);
      if (!compressed_with_header) return NULL;

      // Step 3: Encrypt
      char *encrypted = encrypt(compressed_with_header, *final_compressed_size, output_size, key);
      if (encrypted != compressed_with_header)
  	free(compressed_with_header);

      if (!encrypted || !*output_size) {
          free(encrypted);
          return NULL;
      }
      return encrypted;
  }

  static int build_stage2_buffer(t_arch arch, size_t elf_size, uint64_t s3_vaddr,
  			       char **out_buffer, size_t *out_size,
  			       size_t *payload_raw_size)
  {
      static const size_t stage2_patch_offsets[NUM_ARCHS][2] = {
          [ARCH_X86] = { 0x5B, 0x98 },
          [ARCH_X64] = { 0x6E, 0xD7 }
      };
      const t_type patches[2] = {{.b8 = elf_size}, {.b8 = s3_vaddr}};
      char *stage2_buf;
      size_t stage2_sz;
      char *payload_raw;

      if (load_and_apply_patches_to_buffer(arch, &stage2_buf, &stage2_sz, STAGE_2,
  					 stage2_patch_offsets[arch], patches, 2))
  	return (ERR_LOAD_STAGE);
      if (load_stage_buffer(arch, &payload_raw, payload_raw_size, PAYLOAD_BIN)) {
          free(stage2_buf);
          return (ERR_LOAD_STAGE);
      }
      ,*out_buffer = buffer_join(out_size, payload_raw, *payload_raw_size, stage2_buf, stage2_sz);
      free(payload_raw);
      free(stage2_buf);
      if (!*out_buffer)
  	return (ERR_ALLOC);
      return (ERR_SUCCESS);
  }

   ,*        [Stage1][Encrypted([OriginalSize][LZ77( [Payload][Stage2] + ELF )])]

  int generate_payload(void *elf_mmap, size_t elf_size, char **payload_ptr, size_t *payload_size)
  {
      const unsigned char class = elf_get_elfh_e_ident_class(elf_mmap);
      const t_arch arch = (class == ELFCLASS64) ? ARCH_X64 : ARCH_X86;
      const size_t value_size = arch ? sizeof(uint64_t) : sizeof(uint32_t);
      t_elf_h elf_header = { .elf32 = elf_mmap };
      uint64_t injected_s3_vaddr = 0;

      if (elf_inject_s3(elf_mmap, &elf_header, class, &injected_s3_vaddr) != ERR_SUCCESS)
          return (ERR_INJECT_S3); 
      size_t payload_size_raw;
      size_t combined_len;
      char *combined_payload_stage2;
      if (build_stage2_buffer(arch, elf_size, injected_s3_vaddr, &combined_payload_stage2, &combined_len, &payload_size_raw))
  	return (ERR_LOAD_STAGE);
      if (!combined_payload_stage2)
  	return (ERR_ALLOC);
      size_t full_image_size;
      char *full_image = buffer_join(&full_image_size, combined_payload_stage2,
                                     combined_len, elf_mmap, elf_size);
      free(combined_payload_stage2);
      if (!full_image)
          return (ERR_ALLOC);
      size_t encrypted_size;
      size_t compressed_with_header_size;
      char *key = NULL;
      char *encrypted_payload = compress_encrypt_with_header(
  							   full_image, full_image_size, value_size,
  							   &encrypted_size, &compressed_with_header_size, &key);
      free(full_image);
      if (!encrypted_payload || encrypted_size == 0)
          return (ERR_COMPRESS);

      char *stage1_buffer;
      size_t stage1_size;
      if (build_stage1_patched(class, &elf_header, full_image_size, encrypted_size,
  			     compressed_with_header_size, payload_size_raw,
  			     &stage1_buffer, &stage1_size, key))
  	{
  	    free(encrypted_payload);
  	    return (ERR_LOAD_STAGE);
  	}
      ,*payload_ptr = buffer_join(payload_size, stage1_buffer, stage1_size,
  			       encrypted_payload, encrypted_size);
      free(stage1_buffer);
      free(encrypted_payload);
      if (!*payload_ptr)
          return (ERR_ALLOC);
      return (ERR_SUCCESS);
  }

      while (i && times < num)

      uint64_t p_paddr;
      uint64_t p_type;
      uint64_t p_flags;

      ndx_ph = 0;
      phdr = elf_get_program_header(ehdr, ndx_ph);

      p_align = 0x1000;
      p_filesz = p_align;
      p_memsz = elf_get_highest_used_vaddr_avalaible(ehdr, class, phnum);
      p_offset = 0;
      p_vaddr = elf_get_program_header_vaddr(&phdr, class) & ~(p_align - 1);
      p_memsz -= p_vaddr;
      p_paddr = elf_get_program_header_paddr(&phdr, class) & ~(p_align - 1);
      p_type = PT_LOAD;
      p_flags = PF_R | PF_W;

      elf_phdr_set_paddr(&phdr, p_paddr, class);
      elf_phdr_set_type(&phdr, p_type, class);
      elf_phdr_set_flags(&phdr, p_flags, class);
      elf_phdr_set_align(&phdr, p_align, class);
      if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
  	return (ERR_WRITE_PHDR);
      p_flags = PF_R | PF_X;
      p_vaddr = new_phdr_vaddr;

  }

  int generate_packed_file(void *elf_mmap, void *payload, size_t payload_size, int packed)
  {
      t_elf_h ehdr = (t_elf_h){.elf32 = elf_mmap};
      uint64_t shnum;
      uint64_t phnum;
      unsigned char class;
      uint64_t new_phdr_vaddr;
      uint64_t new_phnum;

      if (validate_elf_headers(&ehdr, &shnum, &phnum, &class))
  	return (ERR_INVALID_ELF);
      if (prepare_info(&ehdr, class, &new_phdr_vaddr, phnum))
  	return (ERR_PREPARE_INFO);
      if (write_ehdr(&ehdr, packed, class, new_phdr_vaddr, &new_phnum, phnum))
  	return (ERR_WRITE_HEADER);
      if (write_program_headers(&ehdr, packed, phnum, new_phdr_vaddr, class, payload_size))
  	return (ERR_WRITE_PHDR);
      if (write_payload(packed, &payload, payload_size))
  	return (ERR_WRITE_PAYLOAD);
      /* ft_printf("Successfully packed\n"); */
      return (0);
  }

  int elf_packer(void *elf_mmap, off_t file_size, const char *file_path)
  {
      size_t payload_size = 0;
      char *payload = NULL;
      if (generate_payload(elf_mmap, (size_t)file_size, &payload, &payload_size))
  	return (5);
      if (!payload || !payload_size)
  	{
  	    free(payload);
  	    return (6);
  	}
      // open the file where we gonna add the payload
      int packed = open(file_path, O_CREAT | O_TRUNC | O_RDWR, S_IRWXU | S_IRWXG | S_IRWXO);
      if (packed < 0)
  	{
  	    free(payload);
  	    return (7);
  	}

      if (generate_packed_file(elf_mmap, payload, payload_size, packed))
  	{
  	    free(payload);
  	    close(packed);
  	    return (8);
  	}
      free(payload);
      close(packed);
      return (0);
  }
