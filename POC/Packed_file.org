* Implementation:
** Scheme view of the struct of our packed file
| ehdr | phdr | s1+crypt_compress_s2+elf_file | end |
- ehdr -> modified {discard all shs data, modify phnum, entry}.
- phdr -> one phdr that use all memsz of og bin, a new one for s1(stage 1), maybe a gnu_stack.
- s1 + crypt_compress_s2 + elf_file
  - s1: get aslr, va, size s1, get fd of /proc/self/exe, mmap s2 RW-, decrypt, decompress in s2, mprotect s2, jmp s2.
  - s2: mmap with fd, close fd, munmap s1, load elf, load interp, set auxv, jmp s3, payload.
  - s3: munmap s2, reset r/edx, jmp to entrypoint(interpe or elf).  if couldn't append to one phdr R-X of the elf file s2 create one.
**  Not update AT_EXECFN (auxv)
The dynamic linker needs the executable path for the AT_EXECFN entry in the auxiliary vector:
    This entry must contain the full path to the executable
    Without it, certain runtime behaviors would fail
   Without preserving the path first, UPX would have no way to set up AT_EXECFN correctly.
The /proc/self/exe Limitation
Special virtual file that:
    Is only available while the executable is mapped in memory
    Disappears if all mappings of the executable are removed
    Cannot be used after complete unmapping
By calling readlink early, UPX captures the actual filesystem path before it unmaps anything.
Already do by OS.

  size_t elf_get_highest_used_vaddr_avalaible(t_elf_h *ehdr, unsigned char class, uint64_t phnum)
  {
      t_elf_ph phdr;
      uint64_t i;
      size_t p_vaddr;
      size_t p_end_vaddr;
      size_t p_memsz;
      size_t highest_used_addr = 0;

      i = 0;
      while (i < phnum)
  	{
  	    phdr = elf_get_program_header(ehdr, i);
  	    if (elf_get_program_header_type(&phdr, class) != PT_LOAD) // If not loadable ignore
  		{
  		    i++;
  		    continue ;
  		}
  	    p_vaddr = elf_get_program_header_vaddr(&phdr, class);
  	    p_memsz = elf_get_program_header_memsz(&phdr, class);
  	    p_end_vaddr = p_vaddr + p_memsz;
  	    if (highest_used_addr <= p_end_vaddr)
  		highest_used_addr = p_end_vaddr;
  	    i++;
  	}
      return  (highest_used_addr);
  }

  static uint64_t elf_get_closest_higher_p_offset(const t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t p_offset)
  {
      t_elf_ph phdr;
      uint64_t ph_ndx;
      uint64_t closest_offset = -1;
      uint64_t cur_offset;
      ph_ndx = 0;

  	{
  	    phdr = elf_get_program_header(ehdr, ph_ndx);
  	    cur_offset = elf_get_program_header_offset(&phdr, class);
  	    if (cur_offset > p_offset && cur_offset < closest_offset)
  		closest_offset = cur_offset;
  	    ph_ndx++;
  	}
      return (closest_offset);
  }

  t_buffer	load_stage_buffer(t_arch arch, t_bin_type bin_type)
  {
      int stage_fd;
      t_buffer stage;

  	ft_bzero(&stage, sizeof(stage));
      stage_fd = open(g_binary_paths[arch][bin_type], O_RDONLY);
      if (stage_fd < 0 || get_file_size(stage_fd, (off_t *)&stage.len) || !stage.len)
      {
  		if (stage_fd >= 0)
  			close(stage_fd);
  		return (stage);
      }
      stage.data = malloc(stage.len);
      if (!stage.data)
      {
  		close(stage_fd);
  		return (stage);
      }
      bool invalid = invalid_read(stage_fd, stage.data, stage.len);
      close(stage_fd);
      if (invalid)
      {
  		free(stage.data);
  		ft_bzero(&stage, sizeof(stage));
      }
      return (stage);
  }

  t_buffer lz77_get_buffer(t_buffer *input)
  {
      t_list *compressed_list;

  t_buffer lz77_get_headed_buffer(t_buffer *input, size_t value_size)
  {
  	t_buffer tmp = {.data=&input->len, .len=value_size};

  	compressed = lz77_get_buffer(input);
  	if (invalid_buffer(&compressed))
  		return (compressed);
  	compressed_with_header = alloc_concat_buffers(&tmp, &compressed);
  	free(compressed.data);
  	return (compressed_with_header);
  }

  {
  	static const size_t stage2_patch_offsets[NUM_ARCHS][2] = {
  		[ARCH_X86] = { 0x5B, 0x98 },
  		[ARCH_X64] = { 0x6E, 0xD7 }

  static t_stage2_result build_stage2_buffer(t_arch arch, size_t elf_size, uint64_t s3_vaddr)
  {
  	const t_patch_set patch = init_patch_stage2(arch, elf_size, s3_vaddr);
  	t_stage2_result out;
  	t_buffer stage2;
  	t_buffer raw_payload;

  	ft_bzero(&out, sizeof(out));
  	stage2 = load_and_apply_patches_to_buffer(arch, STAGE_2, &patch);
  	if (invalid_buffer(&stage2))
  		return (out);
  	raw_payload =load_stage_buffer(arch, PAYLOAD_BIN);
  	if (invalid_buffer(&raw_payload))
  	{
  		free(stage2.data);
  		return (out);
  	}
  	out.payload_raw_size = raw_payload.len;
  	out.buffer = alloc_concat_buffers(&raw_payload, &stage2);
  	free(raw_payload.data);
  	free(stage2.data);
  	return (out);
  }

  static bool elf_set_s3_get_info(const t_elf_h *ehdr, unsigned char class, uint64_t phnum, size_t gap_size, uint64_t *ret_vaddr, uint64_t *ret_offset, t_elf_ph *phdr)
  {
      if (!ehdr || !phnum)
  	return (false);
      if (!gap_size)
  	return (true);
      uint64_t ph_ndx;
      uint64_t cur_vaddr_top;//cur
      uint64_t cur_offset_top;//cur
      uint64_t top_vaddr;
      uint64_t top_offset;//get closest higher offset.
      uint64_t avalaible_vaddr;
      uint64_t avalaible_offset;

      ph_ndx = 0;
      while (ph_ndx < phnum)
  	{
  	    ,*phdr = elf_get_program_header(ehdr, ph_ndx);
  	    if (!(PT_LOAD == elf_get_program_header_type(phdr, class) && PF_X & elf_get_program_header_flags(phdr, class)))
  		{
  		    ph_ndx++;
  		    continue;
  		}
  	    cur_vaddr_top = elf_get_program_header_vaddr(phdr, class) + elf_get_program_header_memsz(phdr, class);
  	    top_vaddr = align_up(cur_vaddr_top, 0x1000);
  	    avalaible_vaddr = top_vaddr - cur_vaddr_top;
  	    if (avalaible_vaddr < gap_size)
  		{
  		    ph_ndx++;
  		    continue;
  		}
  	    cur_offset_top = elf_get_program_header_offset(phdr, class) + elf_get_program_header_memsz(phdr, class);
  	    top_offset = elf_get_closest_higher_p_offset(ehdr, class, phnum, cur_offset_top);
  	    if (top_offset == (uint64_t) -1)
  		top_offset = elf_get_elfh_shoff(ehdr);
  	    avalaible_offset = top_offset - cur_offset_top;
  	    if (avalaible_offset >= gap_size)
  		{
  		    ,*ret_offset = cur_offset_top;
  		    ,*ret_vaddr = cur_vaddr_top;
  		    return (true);
  		}
  	    ph_ndx++;
  	}
      return (false);
  }

  static int elf_inject_s3(void *elf, t_elf_h *ehdr, unsigned char class,
  			 uint64_t *free_va)
  {
      const uint64_t phnum = elf_get_elfh_phnum(ehdr);
      t_elf_ph phdr;
  	t_buffer stage3;
      uint64_t free_off;
      t_arch arch = class == ELFCLASS64;

      stage3 = load_stage_buffer(arch, STAGE_3);
  	if (invalid_buffer(&stage3))
  		return (ERR_LOAD_STAGE);
      //fine
      if (!elf_set_s3_get_info(ehdr, class, phnum, stage3.len, free_va, &free_off, &phdr))
  	{
  	    free(stage3.data);
  	    ,*free_va = 0;
  	    return (ERR_SUCCESS);
  	}
      ft_memcpy(elf + free_off, stage3.data, stage3.len);
      free(stage3.data);
      elf_phdr_set_filesz(&phdr, elf_get_program_header_filesz(&phdr, class) + stage3.len, class);
      elf_phdr_set_memsz(&phdr, elf_get_program_header_memsz(&phdr, class) + stage3.len, class);
      return (ERR_SUCCESS);
  }

  		[ARCH_X64] = { 0x08, 0x1f, 0x60, 0x81, 0xa0, 0xce, 0xce, 0xce, 0xce, 0xce}

  	if (invalid_buffer(&stage1))

  		{.b8 = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum) + prefix_size + 5 + (!arch * 0x72)}, // Calc aslr

  		{.b8 = (uint64_t)params->key}
      };
      t_patch_set patch = ft_init_patch_set(stage1_patch_offsets[arch], stage1_patches, 6, value_size);

  t_stage2_result generate_stage2(t_arch arch, t_buffer *elf)

  	t_stage2_result stage2;

  	uint64_t s3_va;
  	const unsigned char class = elf_get_elfh_e_ident_class(elf->data);

  	elf_header = (t_elf_h){.elf32 = elf->data};
  	ft_bzero(&stage2, sizeof(stage2));

  }


  t_buffer generate_payload(t_buffer *elf)
  {
    const unsigned char class = elf_get_elfh_e_ident_class(elf->data);
    const t_arch arch = (class == ELFCLASS64) ? ARCH_X64 : ARCH_X86;
    const size_t value_size = arch ? sizeof(uint64_t) : sizeof(uint32_t);
    t_elf_h elf_header = (t_elf_h){ .elf32 = elf->data };
    t_compress_encrypt com_cypher;
    t_buffer payload;
    t_stage2_result payload_stage2;
    t_buffer stage1;
    t_buffer payload_stage2_elf;


    ft_bzero(&payload, sizeof(payload));
    payload_stage2 = generate_stage2(arch, elf);
    if (invalid_buffer(&payload_stage2.buffer))
      return (payload);
    payload_stage2_elf = alloc_concat_buffers(&payload_stage2.buffer, elf);
    free(payload_stage2.buffer.data);
    if (invalid_buffer(&payload_stage2_elf))
      return (payload);
    com_cypher = compress_encrypt_with_header(&payload_stage2_elf, value_size);
    free(payload_stage2_elf.data);
    if (invalid_buffer(&com_cypher.encrypt.buffer))
      return (payload);
    t_stage1_params params;

    params.stage2_total_size=payload_stage2_elf.len;
    params.encrypted_size = com_cypher.encrypt.buffer.len;
    params.compressed_hdr_size = com_cypher.head_compress_len;
    params.payload_raw_size = payload_stage2.payload_raw_size;
    params.key = &com_cypher.encrypt.key;

  static int write_ehdr(t_elf_h *ehdr, int packed, unsigned char class,
  		      uint64_t new_phdr_vaddr, uint64_t *new_phnum ,uint64_t phnum)
  {
    ssize_t ehdr_len = elf_get_struct_size_elf_header((char *)ehdr->elf32);

    ,*new_phnum = 2 + (0 != elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum));// if is a PHDR PT_GNU_STACK
    elf_ehdr_set_entry(ehdr, new_phdr_vaddr + ehdr_len + elf_get_elfh_phentsize(ehdr) * *new_phnum, class);
    elf_ehdr_set_phnum(ehdr, *new_phnum, class);

  static int write_program_headers(t_elf_h *ehdr, int packed, uint64_t phnum, uint64_t new_phdr_vaddr,
                                   unsigned char class, uint64_t payload_size)
  {
    const uint64_t ehdr_struct_size = elf_get_struct_size_elf_header((char *)ehdr->elf32);
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    t_elf_ph phdr;
    uint64_t ndx_ph;
    uint64_t p_offset;
    uint64_t p_filesz;
    uint64_t p_memsz;
    uint64_t p_align;
    uint64_t p_vaddr;
    uint64_t p_paddr;
    uint64_t p_type;
    uint64_t p_flags;

    ndx_ph = 0;
    phdr = elf_get_program_header(ehdr, ndx_ph);

    p_align = 0x1000;
    p_filesz = p_align;
    p_memsz = elf_get_highest_used_vaddr_avalaible(ehdr, class, phnum);
    p_offset = 0;
    p_vaddr = elf_get_program_header_vaddr(&phdr, class) & ~(p_align - 1);
    p_memsz -= p_vaddr;
    p_paddr = elf_get_program_header_paddr(&phdr, class) & ~(p_align - 1);
    p_type = PT_LOAD;
    p_flags = PF_R | PF_W;
    elf_phdr_set_filesz(&phdr, p_filesz, class);
    elf_phdr_set_memsz(&phdr, p_memsz, class);
    elf_phdr_set_offset(&phdr, p_offset, class);
    elf_phdr_set_vaddr(&phdr, p_vaddr, class);
    elf_phdr_set_paddr(&phdr, p_paddr, class);
    elf_phdr_set_type(&phdr, p_type, class);
    elf_phdr_set_flags(&phdr, p_flags, class);
    elf_phdr_set_align(&phdr, p_align, class);
    if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      return (ERR_WRITE_PHDR);
    p_flags = PF_R | PF_X;
    p_vaddr = new_phdr_vaddr;
    p_paddr = p_vaddr;
    p_memsz = payload_size + ehdr_struct_size + phdr_struct_size * (2 + (0 != elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum)));

    p_filesz = p_memsz;
    elf_phdr_set_filesz(&phdr, p_filesz, class);
    elf_phdr_set_memsz(&phdr, p_memsz, class);
    elf_phdr_set_vaddr(&phdr, p_vaddr, class);
    elf_phdr_set_paddr(&phdr, p_paddr, class);
    elf_phdr_set_flags(&phdr, p_flags, class);
    if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      return (ERR_WRITE_PHDR);
    ndx_ph = elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum);
    if (ndx_ph)
    {
      phdr = elf_get_program_header(ehdr, ndx_ph);
      if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
        return (ERR_WRITE_PHDR);
    }
    return (0);
  }


  int generate_packed_file(void *elf_mmap, t_buffer *payload, int packed)
  {
      t_elf_h ehdr = (t_elf_h){.elf32 = elf_mmap};
      uint64_t shnum;
      uint64_t phnum;
      unsigned char class;
      uint64_t new_phdr_vaddr;
      uint64_t new_phnum;

      if (validate_elf_headers(&ehdr, &shnum, &phnum, &class))
  		return (ERR_INVALID_ELF);
      new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(&ehdr, class, phnum);
      if (write_ehdr(&ehdr, packed, class, new_phdr_vaddr, &new_phnum, phnum))
  		return (ERR_WRITE_HEADER);
      if (write_program_headers(&ehdr, packed, phnum, new_phdr_vaddr, class, payload->len))
  		return (ERR_WRITE_PHDR);
      if (invalid_write(packed, payload->data, payload->len))
  		return (ERR_WRITE_PAYLOAD);
      /* ft_printf("Successfully packed\n"); */
      return (0);
  }
