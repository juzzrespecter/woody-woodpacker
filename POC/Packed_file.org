  {
    t_elf_ph phdr;
    uint64_t i;
    uint64_t p_vaddr;

    i = 0;
    while (++i < phnum) // Skip the first 
      {
        phdr = elf_get_program_header(ehdr, i);
        p_vaddr = elf_get_program_header_vaddr(&phdr, class);
        //  is loadable phdr
        if (PT_LOAD == elf_get_program_header_type(&phdr, class) && is_between(p_vaddr, p_vaddr + elf_get_program_header_filesz(&phdr, class) - 1, addr))
  	return  (i);
      }
    return  (i);
  }

  size_t set_xor_register_register(void *dst, t_registers reg, unsigned char class)
  {
    const char bytecode_regmap[] = {'\xc0', '\xc0', '\xdb', '\xdb', '\xc9', '\xc9', '\xd2', '\xd2', '\xff', '\xff', '\xf6', '\xf6', '\xed', '\xed', '\xe4', '\xe4', '\xc0', '\xc9', '\xd2','\xdb','\xe4', '\xed', '\xf6', '\xff'};
    size_t bytes;

    if (reg > R15 || (ELFCLASS32 == class && reg >= R8))
      return (-1);
    bytes = 0;
    if (ELFCLASS64 == class)
      {
        if (dst && reg >= R8 && reg <= R15)
  	memcpy(dst + bytes, "\x4D", 1);
        else if (dst)
  	memcpy(dst + bytes, "\x48", 1);
        bytes++;
      }
    if (dst)
      memcpy(dst + bytes, "\x31", 1);
    bytes++;
    if (dst)
      memcpy(dst + bytes, &bytecode_regmap[reg], 1);
    return (++bytes);
  }

  size_t set_xor_register_size(void *dst, unsigned char class)
  {
    size_t xor_register_size;
    t_registers reg;

    if (ELFCLASS64 == class)
      reg = RAX;// RAX gonna hold the value of entrypoint.
    else
      reg = EAX;
    xor_register_size = 0;
    while ((ELFCLASS64 == class && reg <= R15) || reg < ESP)
      {
        if (reg == R13)
  	reg++;
        if (dst)
  	xor_register_size += set_xor_register_register(dst + xor_register_size, reg, class);
        else
  	xor_register_size += set_xor_register_register(NULL, reg, class);
        if (reg >= R8)
  	reg++; 
        else
  	reg += 2;
        if (reg == RSP)//No xor rsp
  	reg += 2;
      }
    return (xor_register_size);
  }

  size_t add_r13(void *dst, uint64_t value)
  {
    size_t written;
    t_type val = (t_type){.b8=value};

    written = 0;
    if (dst)
      memcpy(dst+ written, "\x49\x81\xc5", 3);
    written +=3;
    if (val.b4 == val.b8)
      {
        if (dst)
  	  memcpy(dst + written, &val.b4, sizeof(val.b4));
        written += sizeof(val.b4);
      }
    else
      {
        if (dst)
  	  memcpy(dst + written, &val.b8, sizeof(val.b8));
        written += sizeof(val.b8);
      }
    return (written);
  }


  size_t pop_r13(void *dst)
  {
    if (dst)
      memcpy(dst, "\x41\x5d", 2);
    return (2);
  }

  size_t set_jmp_entrypoint(void *payload, uint64_t entrypoint, unsigned char class)
  {
    t_elf_addr u_entrypoint = (t_elf_addr){.b64 = entrypoint};
    size_t offset = 0;

    if (class == ELFCLASS64)
      {
        /* memcpy(payload, "\x48", 1); //rex prefix x64 */
        if (payload)
  	memcpy(payload, "\x49", 1);
        offset++;
        if (payload)
  	memcpy(payload + offset, "\xbd", 1); //mov r13
        offset++;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b64, sizeof (u_entrypoint.b64));
        offset += sizeof (u_entrypoint.b64);
        if (payload)
  	{
  	offset += pop_r13(payload + offset);
  	offset += add_r13(payload + offset, u_entrypoint.b64);
  	offset += set_jmp_r13(payload + offset);
  	}
        else
  	{
  	offset += pop_r13(NULL);
  	offset += add_r13(NULL, u_entrypoint.b64);
  	offset += set_jmp_r13(NULL);
  	}
      }
    else
      {
        if (payload)
  	memcpy(payload + offset, "\x81\x04\x24", 3);// add DWORD [esp]
        offset+= 3;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b32, sizeof (u_entrypoint.b32));
        offset += sizeof (u_entrypoint.b32);
        if (payload)
  	memcpy(payload + offset, "\xc3", 1);//ret
        offset++;
      }
    return (offset);
  }

  size_t decrypt_rot13_asm(unsigned char *dst, size_t payload_size, unsigned char class)
  {
    size_t written = 0;
    size_t i = 0;

    while (i < payload_size)
      { /* dst + written segfault */
        if (dst)
          written += ft_sub_asm_rsp_buffer_byte(dst + written, i, 13, class);
        else
          written += ft_sub_asm_rsp_buffer_byte(dst, i, 13, class);
        i++;
      }
    return (written);
  }

  size_t set_mov_edi(void *dst, uint32_t value)
  {
    if (dst)
      {
        memcpy(dst, "\xbf", 1);
        memcpy(dst + 1, &value, sizeof(value));
      }
    return (sizeof(value) + 1);
  }

  static size_t copy_payload_to_mmap_aux(unsigned char *dst, size_t offset)
  {
    size_t written;
    written = ft_mov_cl_from_rsp(dst, offset);
    if (dst)
      return (written + ft_mov_rax_from_cl(dst + written, offset));
    return (written + ft_mov_rax_from_cl(dst, offset));
  }

  size_t copy_payload_to_mmap(unsigned char *dst, size_t payload_size)
  {
    size_t written = 0;
    size_t i = 0;

    while (i < payload_size)
      { /* dst + written segfault if dst == NULL*/
        if (dst)
          written += copy_payload_to_mmap_aux(dst + written, i);
        else
          written += copy_payload_to_mmap_aux(dst, i);
        i++;
      }
    return (written);
  }

  static t_elf_ph get_n_loadable_and_exec_phdr(t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t n)

    while (++ph_ndx < phnum)

  static bool increment_offset_phdr(t_elf_h *ehdr, uint64_t offset, uint64_t phnum, unsigned char class)

  uint32_t get_new_e_entrypoint(t_elf_h *ehdr, unsigned char class)
  {
    uint64_t phnum = elf_get_elfh_phnum(ehdr);
    uint64_t shnum = elf_get_elfh_shnum(ehdr);
    uint64_t new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    uint64_t new_phdr_offset = elf_get_highest_offset_avalaible(ehdr, class, shnum);

    const uint64_t new_align = 0x1000;
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    if (increment_offset_phdr(ehdr, new_phdr_offset, phnum, class))// I think shouldn't go throught this way.
      new_phdr_offset += phdr_struct_size;
    if (new_phdr_vaddr % new_align == new_phdr_offset % new_align) 
      printf("all fine with addr and offset aligned\n");
    else
      {
        if (new_phdr_vaddr % new_align > new_phdr_offset % new_align)
  	new_phdr_offset += new_phdr_vaddr % new_align - new_phdr_offset % new_align;
        else
  	new_phdr_vaddr += new_phdr_offset % new_align - new_phdr_vaddr % new_align;
      }
    if (new_phdr_offset % 16)
      {
        new_phdr_vaddr += 16 - new_phdr_vaddr % 16;
        new_phdr_offset += 16 - new_phdr_offset % 16;
      }
    return ((uint32_t)new_phdr_vaddr);
  }

  static int write_program_headers(t_elf_h *ehdr, int packed, uint64_t phnum,
  				 uint64_t new_phdr_offset, uint64_t new_phdr_vaddr,
                                   unsigned char class, uint64_t payload_size, uint64_t *writed)
  {
    t_elf_ph phdr_loadable_and_exec_first = get_n_loadable_and_exec_phdr(ehdr, class, phnum, 1);
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    uint64_t ndx_ph;

    t_elf_ph phdr_cp = {.elf32 = malloc(phdr_struct_size)};
    if (!phdr_cp.elf32 | !phdr_loadable_and_exec_first.elf32)
      return 15;
    ndx_ph = 0;
    t_elf_ph phdr = elf_get_program_header(ehdr, ndx_ph);
    uint64_t p_offset;
    memcpy(phdr_cp.elf32, phdr.elf32, phdr_struct_size);
    elf_phdr_set_filesz(&phdr_cp, elf_get_program_header_filesz(&phdr, class) + phdr_struct_size, class);
    elf_phdr_set_memsz(&phdr_cp, elf_get_program_header_memsz(&phdr, class) + phdr_struct_size, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      return (17);
    ,*writed += phdr_struct_size;
    //Write modify the phdr already exist
    while (++ndx_ph < phnum)
      {
        phdr = elf_get_program_header(ehdr, ndx_ph);
        memcpy(phdr_cp.elf32, phdr.elf32, phdr_struct_size);

        p_offset = elf_get_program_header_offset(&phdr, class);
        if (increment_offset_phdr(ehdr, p_offset, phnum, class))//phdr after > ehdr and before 2 load offset
  	{
  	  elf_phdr_set_offset(&phdr_cp, p_offset + phdr_struct_size, class);
  	  if (elf_get_program_header_vaddr(&phdr, class))
  	    elf_phdr_set_vaddr(&phdr_cp, elf_get_program_header_vaddr(&phdr, class) + phdr_struct_size, class);
  	  if (elf_get_program_header_paddr(&phdr, class)) // BSD
  	    elf_phdr_set_paddr(&phdr_cp, elf_get_program_header_paddr(&phdr, class) + phdr_struct_size, class);
  	}
        else if (!p_offset && PT_LOAD == elf_get_program_header_type(&phdr, class))//first load include last phdr.
  	{
  	  elf_phdr_set_filesz(&phdr_cp, elf_get_program_header_filesz(&phdr, class) + phdr_struct_size, class);
  	  elf_phdr_set_memsz(&phdr_cp, elf_get_program_header_memsz(&phdr, class) + phdr_struct_size, class);
  	  if (elf_get_program_header_memsz(&phdr_cp, class) + elf_get_program_header_vaddr(&phdr_cp, class) > elf_get_program_header_vaddr(&phdr_loadable_and_exec_first, class))
  	    return (16);// Overlap the address of the next phdr.
  	}
        if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
  	return (17);
        ,*writed += phdr_struct_size;
      }
    //Set the new_phdr.
    memcpy(phdr_cp.elf32, phdr_loadable_and_exec_first.elf32, phdr_struct_size);// phdr .text
    //Modify memsz, filesz

    elf_phdr_set_memsz(&phdr_cp, payload_size, class);
    elf_phdr_set_filesz(&phdr_cp, payload_size, class);
    elf_phdr_set_vaddr(&phdr_cp, new_phdr_vaddr, class);
    elf_phdr_set_flags(&phdr_cp, PF_X | PF_R, class);// May now don't need but don't care.
    if (elf_get_program_header_paddr(&phdr_cp, class))// Under BSD Is unused and must be Zero
      elf_phdr_set_paddr(&phdr_cp, new_phdr_vaddr, class);
    elf_phdr_set_offset(&phdr_cp, new_phdr_offset, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      return (18);
    ,*writed += phdr_struct_size;
    free(phdr_cp.elf32);
    return 0;
  }

  static int write_padding_and_payload(int packed, void **generated_bytecode, uint64_t payload_size,
  				     uint64_t padding, uint64_t *writed)
  {
    printf("padding: %lu\n", padding);
    while (padding--)
      {
        if (write(packed, "\0", 1) != 1)
  	return (1);
        writed[0]++;
      }
    if (write(packed, *generated_bytecode, payload_size) != (ssize_t)payload_size)
      {
        free(*generated_bytecode);
        ,*generated_bytecode = NULL;
        return (2);
      }
    free(*generated_bytecode);
    ,*generated_bytecode = NULL;
    ,*writed += payload_size;
    return (0);
  }

  static int write_remaining_section_headers(t_elf_h *ehdr, int packed, uint64_t phnum, uint64_t shnum,
                                             unsigned char class, uint64_t new_phdr_offset,
                                             uint64_t new_phdr_vaddr, uint64_t *writed, uint64_t payload_size)
  {
    const uint64_t alignment = 16;
    uint64_t shdr_struct_size = elf_get_section_header_struct_size(ehdr);
    uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    t_elf_sh shdr_cp = (t_elf_sh){.elf32 = malloc(shdr_struct_size)};
    uint64_t ndx_sh;
    if (!shdr_cp.elf32)
      return (12);
    printf("writed %lu\n", *writed);
    uint64_t padding_size = alignment - *writed % alignment;
    printf("padding to begin Shdr: %lu\n", padding_size);

    if (padding_size != alignment)
      {
        if (padding(packed, padding_size))
  	return (1);
        writed[0] += padding_size;
      }
    ndx_sh = 0;
    t_elf_sh shdr = elf_get_section_header(ehdr, ndx_sh);
    memcpy(shdr_cp.elf32, shdr.elf32, shdr_struct_size);
    printf("writed payload begining shdr writed: %ld\n", *writed);
    uint64_t shcp_offset;   
   
    if (write(packed, shdr_cp.elf32, shdr_struct_size) != (ssize_t)shdr_struct_size)
      return (2);
    ,*writed += shdr_struct_size;
    while (++ndx_sh < shnum)
      {
        shdr = elf_get_section_header(ehdr, ndx_sh);
        shcp_offset = elf_get_section_header_offset(&shdr, class);
        memcpy(shdr_cp.elf32, shdr.elf32, shdr_struct_size);
        if (increment_offset_phdr(ehdr, shcp_offset, phnum, class))
  	{
  	  elf_shdr_set_offset(&shdr_cp, elf_get_section_header_offset(&shdr, class) + phdr_struct_size, class);
  	  if (elf_get_section_header_addr(&shdr, class))
  	    elf_shdr_set_addr(&shdr_cp, elf_get_section_header_addr(&shdr, class) + phdr_struct_size, class);
  	}
        if (write(packed, shdr_cp.elf32, shdr_struct_size) != (ssize_t)shdr_struct_size)
  	return (1);
        ,*writed += shdr_struct_size;
      }
    //Modify  flags, name, type, address, offset, align, link, Info, Entsize
    elf_shdr_set_addr(&shdr_cp, new_phdr_vaddr, class);
    elf_shdr_set_offset(&shdr_cp, new_phdr_offset, class);
    elf_shdr_set_size(&shdr_cp, payload_size, class);
    elf_shdr_set_addralign(&shdr_cp, 16, class);
    elf_shdr_set_name(&shdr_cp, elf_get_section_header_name(&shdr_cp, class) + 1, class);
    elf_shdr_set_type(&shdr_cp, SHT_PROGBITS, class);
    elf_shdr_set_flags(&shdr_cp, SHF_ALLOC | SHF_EXECINSTR, class);
    elf_shdr_set_link(&shdr_cp, 0, class);
    elf_shdr_set_info(&shdr_cp, 0, class);
    elf_shdr_set_entsize(&shdr_cp, 0, class);
    ,*writed += write(packed, shdr_cp.elf32, shdr_struct_size);
    free(shdr_cp.elf32);
    return 0;
  }

  int write_dyn(t_elf_h *ehdr, t_elf_sh *shdr, unsigned char class, uint32_t phnum, size_t *writed, int packed)
  {
    if (!shdr->elf32)
      return (1);
    uint64_t dyn_offset = elf_get_section_header_offset(shdr, class);
    uint64_t dyn_size = elf_get_section_header_size(shdr, class);
    uint64_t dyn_entsize = elf_get_struct_size_elf_dyn(class);
              
    t_elf_dyn dyn_cp = {.elf32=malloc(dyn_entsize)};
    if  (!dyn_cp.elf32)
      return (2);
    // Read the dynamic section data
    void *dynamic_data = (void *)((char *)ehdr->elf32 + dyn_offset);
    uint64_t entry_count = dyn_size / dyn_entsize;
    printf("entry_count %lu\n", entry_count); 
    printf("dyn_entsize %lu\n", dyn_entsize); 
    // Update entries that point to shifted addresses
    uint64_t ndx_dyn = 0;
    uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    uint64_t offset;


    while (ndx_dyn < entry_count)
      {
        t_elf_dyn dyn = elf_get_dyn(dynamic_data, class, ndx_dyn);
        memcpy(dyn_cp.elf32, dyn.elf32, dyn_entsize);
        uint64_t d_tag = elf_get_dyn_tag(&dyn, class);
                  
        // Check if this entry contains a virtual address that needs updating
        if (elf_is_addres_type_entry(d_tag))
  	{ 
  	uint64_t d_val = elf_get_dyn_val(&dyn, class);
                      
  	// If this address is in the affected range, update it
  	//Change vaddr for offset.
  	if (vaddr_to_offset_safe(ehdr, d_val, &offset, class))
  	  return (4);
  	if (increment_offset_phdr(ehdr, offset, phnum, class))
  	  {
  	    printf("Updating DT_%ld from 0x%lx to 0x%lx\n", 
  		   d_tag, d_val, d_val + phdr_struct_size);
  	    elf_set_dyn_val(&dyn_cp, d_val + phdr_struct_size, class);
  	  }
        }
        if ((ssize_t) dyn_entsize != write(packed, dyn_cp.elf32, dyn_entsize))
  	return (3);
        ,*writed += dyn_entsize;
        ndx_dyn++;
      }
      return 0;
  }

  int main(int argc, char **argv)
  {
    //validation params
    if (argc != 2)
      {
        printf("usage: %s [filename]\n", *argv);
        return (1);
      }  	
    //open file
    int fd = open(argv[1], O_RDONLY); 
    if (fd < 0)
      return (2);
    // get size file
    const off_t file_size = lseek(fd, 0, SEEK_END);
    if ((off_t) -1 == file_size || (off_t) -1 == lseek(fd, 0, SEEK_SET))
      {
        close(fd);
        return (3);
      }
    // save on memory all the file
    void *elf_mmap = mmap(NULL, file_size, PROT_READ, MAP_SHARED, fd, 0);
    close(fd);
    if (!elf_mmap)
      return (4);
    // open the file where we gonna add the payload
    int packed = open("woody", O_CREAT | O_TRUNC | O_RDWR, S_IRWXU | S_IRWXG | S_IRWXO);
    if (packed < 0)
      {
        munmap(elf_mmap, file_size);
        return (5);
      } // starting poc part
    size_t generated_bytecode_size;
    char *generated_bytecode = NULL;
    if (generate_payload(elf_mmap, &generated_bytecode, &generated_bytecode_size))
      return (6);
    if (!generated_bytecode || !generated_bytecode_size)
      return (7);
    if (generated_packed_file(elf_mmap, file_size, generated_bytecode, generated_bytecode_size, packed))
      return (8);
    close(packed);
    munmap(elf_mmap, file_size);
    return (0);
  }
