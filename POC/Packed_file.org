* Implementation:
** Scheme view of the struct of our packed file
| ehdr | phdr | s1+crypt_compress_s2+elf_file | end |
- ehdr -> modified {discard all shs data, modify phnum, entry}.
- phdr -> one phdr that use all memsz of og bin, a new one for s1(stage 1), maybe a gnu_stack.
- s1 + crypt_compress_s2 + elf_file
  - s1: get aslr, va, size s1, get fd of /proc/self/exe, mmap s2 RW-, decrypt, decompress in s2, mprotect s2, jmp s2.
  - s2: mmap with fd, close fd, munmap s1, load elf, load interp, set auxv, jmp s3, payload.
  - s3: munmap s2, reset r/edx, jmp to entrypoint(interpe or elf).  if couldn't append to one phdr R-X of the elf file s2 create one.
**  Not update AT_EXECFN (auxv)
The dynamic linker needs the executable path for the AT_EXECFN entry in the auxiliary vector:
    This entry must contain the full path to the executable
    Without it, certain runtime behaviors would fail
   Without preserving the path first, UPX would have no way to set up AT_EXECFN correctly.
The /proc/self/exe Limitation
Special virtual file that:
    Is only available while the executable is mapped in memory
    Disappears if all mappings of the executable are removed
    Cannot be used after complete unmapping
By calling readlink early, UPX captures the actual filesystem path before it unmaps anything.
Already do by OS.

  size_t elf_get_highest_used_vaddr_avalaible(t_elf_h *ehdr, unsigned char class, uint64_t phnum)
  {
      t_elf_ph phdr;
      uint64_t i;
      size_t p_vaddr;
      size_t p_end_vaddr;
      size_t p_memsz;
      size_t highest_used_addr = 0;

      i = 0;
      while (i < phnum)
  	{
  	    phdr = elf_get_program_header(ehdr, i);
  	    if (elf_get_program_header_type(&phdr, class) != PT_LOAD) // If not loadable ignore
  		{
  		    i++;
  		    continue ;
  		}
  	    p_vaddr = elf_get_program_header_vaddr(&phdr, class);
  	    p_memsz = elf_get_program_header_memsz(&phdr, class);
  	    p_end_vaddr = p_vaddr + p_memsz;
  	    if (highest_used_addr <= p_end_vaddr)
  		highest_used_addr = p_end_vaddr;
  	    i++;
  	}
      return  (highest_used_addr);
  }

  int load_stage_buffer(t_arch arch, char **stage_buffer, size_t *stage_fd_size, t_bin_type bin_type)
  {
      int stage_fd;
      t_buffer stage;

      // Validate input parameters
      if (!stage_buffer || !stage_fd_size)
  	return (1);
      // Initialize output parameters
      ,*stage_buffer = NULL;
      ,*stage_fd_size = 0;
      stage_fd = open(g_binary_paths[arch][bin_type], O_RDONLY);
      if (stage_fd < 0 || get_file_size(stage_fd, (off_t *)&stage.len) || !stage.len)
      {
  	if (stage_fd >= 0)
  	    close(stage_fd);
  	return (1);
      }
      stage.data = malloc(stage.len);
      if (!stage.data)
      {
  	close(stage_fd);
  	return (5);
      }

      ,*stage_fd_size = stage.len;
      ,*stage_buffer = stage.data;
      bool invalid = invalid_read(stage_fd, stage.data, stage.len);
      close(stage_fd);
      if (invalid)
      {
  	free(stage.data);
  	,*stage_buffer = NULL;
  	close(stage_fd);
  	return (5);
      }
      return (0);
  }

  static int load_and_apply_patches_to_buffer(t_arch arch, char **loaded_buffer, size_t *loaded_size,
  					    t_bin_type bin_type, const size_t *patch_offsets, const t_type *patch_values, size_t patch_count)
  {
    t_buffer binary;

    if (load_stage_buffer(arch, (char **)&binary.data, &binary.len, bin_type))
      {
        ,*loaded_buffer = NULL;
        ,*loaded_size = 0;
        return (5); // Load failed
      }
    ,*loaded_buffer = binary.data;
    ,*loaded_size = binary.len;
    // Step 2: Determine data size for patching (32 vs 64-bit pointers/values)
    size_t value_size = (size_t)ft_ternary(arch, (void *)sizeof(uint64_t), (void *)sizeof(uint32_t));
    // Step 3: Apply all patches
    apply_patches_to_buffer(binary.data, patch_offsets, patch_values, patch_count, value_size);
    return 0; // Success
  }

  //Delete output_size and change return type value for t_buffer.
  static char *compress_encrypt_with_header(const void *input_data, size_t input_size,//Change with t_buffer
  					  size_t value_size, size_t *output_size,					  size_t *final_compressed_size, char **key)
  {
      t_buffer input;

      input.len=input_size;
      input.data=(void *)input_data;
      // Step 1: Compress
      t_list *compressed_list = lz77_compress_to_list((unsigned char *)input.data, input.len);
      if (!compressed_list)
  	return NULL;
      t_buffer compressed;

      compressed.data = lz77_get_compress_arr(compressed_list, &compressed.len);
      ft_lstclear(&compressed_list, free);
      if (!compressed.data)
  	return NULL;
      // Step 2: Prepend original size as header
      t_buffer compressed_with_header;
      compressed_with_header.data = alloc_concat_buffers(&compressed_with_header.len,
  						       &input.len, value_size,
  						       compressed.data, compressed.len);
      ,*final_compressed_size = compressed_with_header.len;
      free(compressed.data);
      if (!compressed_with_header.data)
  	return NULL;

      // Step 3: Encrypt
      t_buffer encrypted;
      encrypted.data = encrypt(compressed_with_header.data, compressed_with_header.len, &encrypted.len, key);

      if (encrypted.data != compressed_with_header.data)
  	free(compressed_with_header.data);
      ,*output_size = encrypted.len;
      if (!encrypted.data || !encrypted.len)
      {
  	free(encrypted.data);
  	return NULL;
      }
      return (encrypted.data);
  }

  static int build_stage2_buffer(t_arch arch, size_t elf_size, uint64_t s3_vaddr,
  			       char **out_buffer, size_t *out_size,
  			       size_t *payload_raw_size)
  {
    static const size_t stage2_patch_offsets[NUM_ARCHS][2] = {
      [ARCH_X86] = { 0x5B, 0x98 },
      [ARCH_X64] = { 0x6E, 0xD7 }
    };
    const t_type patches[2] = {{.b8 = elf_size}, {.b8 = s3_vaddr}};
    t_buffer stage2;
    t_buffer raw_payload;
    if (load_and_apply_patches_to_buffer(arch, (char **)&stage2.data, &stage2.len, STAGE_2,
  				       stage2_patch_offsets[arch], patches, 2))
      return (ERR_LOAD_STAGE);

    if (load_stage_buffer(arch, (char **)&raw_payload.data, &raw_payload.len, PAYLOAD_BIN)) {
      free(stage2.data);
      return (ERR_LOAD_STAGE);
    }
    ,*payload_raw_size = raw_payload.len;
    t_buffer out;
    out.data = alloc_concat_buffers(&out.len, raw_payload.data, raw_payload.len, stage2.data, stage2.len);
    free(raw_payload.data);
    free(stage2.data);
    if (!out.data)
      return (ERR_ALLOC);
    ,*out_buffer = out.data;
    ,*out_size = out.len;
    return (ERR_SUCCESS);
  }

  static bool elf_set_s3_get_info(const t_elf_h *ehdr, unsigned char class, uint64_t phnum, size_t gap_size, uint64_t *ret_vaddr, uint64_t *ret_offset, t_elf_ph *phdr)
  {
      if (!ehdr || !phnum)
  	return (false);
      if (!gap_size)
  	return (true);
      uint64_t ph_ndx;
      uint64_t cur_vaddr_top;//cur
      uint64_t cur_offset_top;//cur
      uint64_t top_vaddr;
      uint64_t top_offset;//get closest higher offset.
      uint64_t avalaible_vaddr;
      uint64_t avalaible_offset;

      ph_ndx = 0;
      while (ph_ndx < phnum)
  	{
  	    ,*phdr = elf_get_program_header(ehdr, ph_ndx);
  	    if (!(PT_LOAD == elf_get_program_header_type(phdr, class) && PF_X & elf_get_program_header_flags(phdr, class)))
  		{
  		    ph_ndx++;
  		    continue;
  		}
  	    cur_vaddr_top = elf_get_program_header_vaddr(phdr, class) + elf_get_program_header_memsz(phdr, class);
  	    top_vaddr = align_up(cur_vaddr_top, 0x1000);
  	    avalaible_vaddr = top_vaddr - cur_vaddr_top;
  	    if (avalaible_vaddr < gap_size)
  		{
  		    ph_ndx++;
  		    continue;
  		}
  	    cur_offset_top = elf_get_program_header_offset(phdr, class) + elf_get_program_header_memsz(phdr, class);
  	    top_offset = elf_get_closest_higher_p_offset(ehdr, class, phnum, cur_offset_top);
  	    if (top_offset == (uint64_t) -1)
  		top_offset = elf_get_elfh_shoff(ehdr);
  	    avalaible_offset = top_offset - cur_offset_top;
  	    if (avalaible_offset >= gap_size)
  		{
  		    ,*ret_offset = cur_offset_top;
  		    ,*ret_vaddr = cur_vaddr_top;
  		    return (true);
  		}
  	    ph_ndx++;
  	}
      return (false);
  }

  int generate_payload(void *elf_mmap, size_t elf_size, char **payload_ptr, size_t *payload_size)
  {
    const unsigned char class = elf_get_elfh_e_ident_class(elf_mmap);
    const t_arch arch = (class == ELFCLASS64) ? ARCH_X64 : ARCH_X86;
    const size_t value_size = arch ? sizeof(uint64_t) : sizeof(uint32_t);
    t_elf_h elf_header = { .elf32 = elf_mmap };
    uint64_t injected_s3_vaddr = 0;

    if (elf_inject_s3(elf_mmap, &elf_header, class, &injected_s3_vaddr) != ERR_SUCCESS)
      return (ERR_INJECT_S3);
    size_t payload_size_raw;
    /* size_t combined_len; */
    /* char *combined_payload_stage2; */
    t_buffer payload_stage2;
    if (build_stage2_buffer(arch, elf_size, injected_s3_vaddr, (char **)&payload_stage2.data, &payload_stage2.len, &payload_size_raw))
      return (ERR_LOAD_STAGE);
    if (!payload_stage2.data)
      return (ERR_ALLOC);
    size_t full_image_size;
    char *full_image = alloc_concat_buffers(&full_image_size, payload_stage2.data,
  				 payload_stage2.len, elf_mmap, elf_size);
    free(payload_stage2.data);
    if (!full_image)
      return (ERR_ALLOC);
    size_t encrypted_size;
    size_t compressed_with_header_size;
    char *key = NULL;
    char *encrypted_payload = compress_encrypt_with_header(full_image, full_image_size, value_size,
  							 &encrypted_size, &compressed_with_header_size, &key);
    free(full_image);
    if (!encrypted_payload || encrypted_size == 0)
      return (ERR_COMPRESS);

    char *stage1_buffer;
    size_t stage1_size;
    if (build_stage1_patched(class, &elf_header, full_image_size, encrypted_size,
  			   compressed_with_header_size, payload_size_raw,
  			   &stage1_buffer, &stage1_size, key))
      {
        free(encrypted_payload);
        return (ERR_LOAD_STAGE);
      }
    ,*payload_ptr = alloc_concat_buffers(payload_size, stage1_buffer, stage1_size,
  			     encrypted_payload, encrypted_size);
    free(stage1_buffer);
    free(encrypted_payload);
    if (!*payload_ptr)
      return (ERR_ALLOC);
    return (ERR_SUCCESS);
  }

    ,*new_phnum = 2 + (0 != elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum));// if is a PHDR PT_GNU_STACK

    elf_ehdr_set_phnum(ehdr, *new_phnum, class);
    elf_ehdr_set_shoff(ehdr, 0, class);
    elf_ehdr_set_shnum(ehdr, 0, class);
    elf_ehdr_set_shentsize(ehdr, 0, class);
    elf_ehdr_set_shstrndx(ehdr, 0, class);
    if (write(packed, ehdr->elf32, ehdr_len) != ehdr_len)
      return (ERR_WRITE_HEADER);
    return (ERR_SUCCESS);
  }

  static int write_program_headers(t_elf_h *ehdr, int packed, uint64_t phnum, uint64_t new_phdr_vaddr,
                                   unsigned char class, uint64_t payload_size)
  {
    const uint64_t ehdr_struct_size = elf_get_struct_size_elf_header((char *)ehdr->elf32);
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    t_elf_ph phdr;
    uint64_t ndx_ph;
    uint64_t p_offset;
    uint64_t p_filesz;
    uint64_t p_memsz;
    uint64_t p_align;
    uint64_t p_vaddr;
    uint64_t p_paddr;
    uint64_t p_type;
    uint64_t p_flags;

    ndx_ph = 0;
    phdr = elf_get_program_header(ehdr, ndx_ph);

    p_align = 0x1000;
    p_filesz = p_align;
    p_memsz = elf_get_highest_used_vaddr_avalaible(ehdr, class, phnum);
    p_offset = 0;
    p_vaddr = elf_get_program_header_vaddr(&phdr, class) & ~(p_align - 1);
    p_memsz -= p_vaddr;
    p_paddr = elf_get_program_header_paddr(&phdr, class) & ~(p_align - 1);
    p_type = PT_LOAD;
    p_flags = PF_R | PF_W;
    elf_phdr_set_filesz(&phdr, p_filesz, class);
    elf_phdr_set_memsz(&phdr, p_memsz, class);
    elf_phdr_set_offset(&phdr, p_offset, class);
    elf_phdr_set_vaddr(&phdr, p_vaddr, class);
    elf_phdr_set_paddr(&phdr, p_paddr, class);
    elf_phdr_set_type(&phdr, p_type, class);
    elf_phdr_set_flags(&phdr, p_flags, class);
    elf_phdr_set_align(&phdr, p_align, class);
    if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      return (ERR_WRITE_PHDR);
    p_flags = PF_R | PF_X;
    p_vaddr = new_phdr_vaddr;
    p_paddr = p_vaddr;
    p_memsz = payload_size + ehdr_struct_size + phdr_struct_size * (2 + (0 != elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum)));

    p_filesz = p_memsz;
    elf_phdr_set_filesz(&phdr, p_filesz, class);
    elf_phdr_set_memsz(&phdr, p_memsz, class);
    elf_phdr_set_vaddr(&phdr, p_vaddr, class);
    elf_phdr_set_paddr(&phdr, p_paddr, class);
    elf_phdr_set_flags(&phdr, p_flags, class);
    if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      return (ERR_WRITE_PHDR);
    ndx_ph = elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum);
    if (ndx_ph)
    {
      phdr = elf_get_program_header(ehdr, ndx_ph);
      if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
        return (ERR_WRITE_PHDR);
    }
    return (0);
  }


  int generate_packed_file(void *elf_mmap, void *payload, size_t payload_size, int packed)
  {
      t_elf_h ehdr = (t_elf_h){.elf32 = elf_mmap};
      uint64_t shnum;
      uint64_t phnum;
      unsigned char class;
      uint64_t new_phdr_vaddr;
      uint64_t new_phnum;

      if (validate_elf_headers(&ehdr, &shnum, &phnum, &class))
  	return (ERR_INVALID_ELF);
      new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(&ehdr, class, phnum);
      if (write_ehdr(&ehdr, packed, class, new_phdr_vaddr, &new_phnum, phnum))
  	return (ERR_WRITE_HEADER);
      if (write_program_headers(&ehdr, packed, phnum, new_phdr_vaddr, class, payload_size))
  	return (ERR_WRITE_PHDR);
      if (write_payload(packed, &payload, payload_size))
  	return (ERR_WRITE_PAYLOAD);
      /* ft_printf("Successfully packed\n"); */
      return (0);
  }

      if (packed < 0)
  	{
  	    free(payload);
  	    return (7);

  	{
  	    free(payload);
  	    close(packed);
  	    return (8);
  	}
      free(payload);
      close(packed);
      return (0);
  }
