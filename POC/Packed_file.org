  t_list	*ft_lstnew(void *content)
  {
  	t_list	*node;

  	node = malloc(sizeof(t_list));
  	if (!node)
  		return (NULL);
  	node->content = content;
  	node->next = NULL;
  	return (node);
  }

  void	ft_lstdelone(t_list *lst, void (*del)(void *))
  {
  	if (!lst || !del)
  		return ;
  	del(lst->content);
  	free(lst);
  }

  void	ft_lstclear(t_list **lst, void (*del)(void *))
  {
  	t_list	*node;
  	t_list	*tmp;

  	if (!lst || !*lst)
  		return ;
  	node = *lst;
  	while (node)
  	{
  		tmp = node->next;
  		ft_lstdelone(node, del);
  		node = tmp;
  	}
  	,*lst = NULL;
  }

  t_list	*ft_lstlast(t_list *lst)
  {
  	if (!lst)
  		return (NULL);
  	while (lst->next)
  		lst = lst->next;
  	return (lst);
  }

  void	ft_lstadd_back(t_list **lst, t_list *new)
  {
  	t_list	*lst_last;

  	if (!new || !lst)
  		return ;
  	if (*lst == NULL)
  		,*lst = new;
  	else
  	{
  		lst_last = ft_lstlast(*lst);
  		lst_last->next = new;
  	}
  }

  int	ft_lstsize(t_list *lst)
  {
  	int	lst_num;

  	if (!(size_t)lst)
  		return (0);
  	lst_num = 1;
  	while (lst->next)
  	{
  		lst_num++;
  		lst = lst->next;

  	return (lst_num);

  t_list *lz77_compress(const unsigned char *data, size_t data_len)
  {
      t_list *lst;
      t_list *tmp;
      t_lz77 *token;

      lst = NULL;
      if (!data || data_len == 0) return (NULL);

      size_t i = 0;
      while (i < data_len)
        {
  	size_t best_offset = 0;
  	size_t best_length = 0;

  	// Search in sliding window: [max(0, i - WINDOW_SIZE), i)
    	size_t window_start = 0;//(i > COMPRESS_LZ77_SLIDING_WINDOW_SIZE) * (i - COMPRESS_LZ77_SLIDING_WINDOW_SIZE);
    	if (i > COMPRESS_LZ77_SLIDING_WINDOW_SIZE)
    	  window_start = i - COMPRESS_LZ77_SLIDING_WINDOW_SIZE;

  	// Try all possible offsets (1 to window_size)
          for (size_t offset = 1; offset <= (i - window_start); offset++) {
  	  size_t match_length = 0;
  	  size_t max_possible = (data_len - i > COMPRESS_LZ77_MAX_MATCH_LEN) ? COMPRESS_LZ77_MAX_MATCH_LEN : (data_len - i);
  	  while (match_length < max_possible &&
  		 data[i - offset + match_length] == data[i + match_length]) {
  	    match_length++;
  	  }
  	  if (match_length >= COMPRESS_LZ77_MIN_MATCH_LEN && match_length > best_length) {
  	    best_length = match_length;
  	    best_offset = offset;
  	  }
  	}
  	uint8_t next_char = 0;
  	if (i + best_length < data_len) {
  	  next_char = data[i + best_length];
  	}

  	token = malloc(sizeof(*token));
  	if (!token)
  	  {
  	    ft_lstclear(&lst, free);
  	    return (NULL);
  	  }

  	if (best_length >= COMPRESS_LZ77_MIN_MATCH_LEN) {
  	  // Valid match: store offset and len-2
  	  if (best_offset > 4095) {
  	    fprintf(stderr, "Offset %zu too large\n", best_offset);
  	    ft_lstclear(&lst, free);
  	    return (NULL);
  	  }
  	  size_t token_len = best_length - 2; // 0, 3 to 17
  	  token->distance_len = (best_offset << 4) | (token_len & 0x0F);
  	  token->byte = next_char;
  	  i += best_length + 1; // Skip matched bytes + literal
  	} else {
  	  // Literal only
  	  token->distance_len = 0; // offset=0, len=0
  	  token->byte = data[i];
  	  i += 1;
  	}

  	// Write token (3 bytes)
  	tmp = ft_lstnew(token);
  	if (!tmp)
  	  {
  	    ft_lstclear(&lst, free);
  	    return (NULL);
  	  }
  	ft_lstadd_back(&lst, tmp);
    	printf("Token: distlen=0x%04x (offset=%zu, len=%zu), byte=0x%02x at i=%zu\n",
    	       token->distance_len,
    	       (size_t)(token->distance_len >> 4),
    	       (size_t)((token->distance_len & 0xF) == 0 ? 0 : ((token->distance_len & 0xF) + 2)),
    	       token->byte,
    	       i - (best_length >= 3 ? best_length + 1 : 1));
        }
      return (lst);
  }

  size_t ft_mov_asm_rsp(void *dst, size_t offset, size_t value) //mov the dst pointer to not overwrite.
  {
    size_t	written;
    const t_type	type = (t_type){.b8 = value};
    const t_type	off = (t_type){.b8 = offset};

    written = 0;
    if (dst)
      memcpy(dst, "\xc6", 1);
    written++;
    if (!off.b8)
      {
        if (dst)
          memcpy(dst + written, "\x04", 1);
      }
    else if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80) // if offset and byte syze
      {
        if (dst)
          memcpy(dst + written, "\x44", 1);
      }
    else // if offset and > byte size
      {
        if (dst)
          memcpy(dst + written, "\x84", 1);
      }
    written++;
    if (dst)
      memcpy(dst + written, "\x24", 1);
    written++;
    if (off.b8)
      {
        /* if (dst) */
        /* printf("off -> b value %u, b4 value %u, b8 value %lu\n", (uint32_t)off.b, off.b4, off.b8); */
        if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80)
          {
            if (dst)
              memcpy(dst + written, &off.b, sizeof (off.b)); 
            written += sizeof (off.b);
          }
        else if ((uint64_t)off.b4 == off.b8 && off.b4 < 0x80000000)
          {
            if (dst)
              memcpy(dst + written, &off.b4, sizeof (off.b4)); 
            written += sizeof (off.b4);
          }
        else
          {
            if (dst)
              memcpy(dst + written, &off.b8, sizeof (off.b8)); 
            written += sizeof (off.b8);
          }
      }
    if ((uint32_t)type.b  == type.b4 && (uint64_t)type.b4 == type.b8)
      {
        if (dst)
          memcpy(dst + written, &type.b, sizeof (type.b));
        written += sizeof (type.b);
      }
    else if ((uint64_t)type.b4 == type.b8 && type.b4)
      {
        if (dst)
          memcpy(dst + written, &type.b4, sizeof (type.b4));
        written += sizeof (type.b4);
      }
    else
      {
        if (dst)
          memcpy(dst + written, &type.b8, sizeof (type.b8));
        written += sizeof (type.b8);
      }
    return (written);
  }

        return (written + sizeof (off.b));
      }
    if (dst)
      memcpy(dst + written, "\x88", 1);
    written++;
    if ((uint64_t)off.b4 == off.b8 && off.b4 < 0x80000000)
      {
        if (dst)
  	memcpy(dst + written, &off.b4, sizeof (off.b4)); 
        return (written + sizeof (off.b4));
      }
    if (dst)
      memcpy(dst + written, &off.b8, sizeof (off.b8)); 
    return (written + sizeof (off.b8));
  }

  size_t ft_sub_asm_rsp(void *dst, uint64_t value, unsigned char class)//mov the dst pointer to not overwrite.
  {
    size_t	written;
    const t_type	type = (t_type){.b8 = value};

    if (ELFCLASS32 == class && type.b8 != type.b4)
      return (-1);
    written = 0;
    if (class == ELFCLASS64)
      {
        if (dst)
  	memcpy(dst, "\x48", 1);
        written++;
      }
    if ((uint32_t)type.b  == type.b4 && (uint64_t)type.b4 == type.b8 && type.b < 0x80)
      {
        if (dst)
  	memcpy(dst + written, "\x83", 1);
        written++;
        if (dst)
  	{
  	  /* if (ELFCLASS64 == class) */
  	    memcpy(dst+written, "\xec", 1);
  	  /* else */
  	    /* memcpy(dst+written, "\xe8", 1); */
  	}
        written++;
        if (dst)
  	memcpy(dst + written, &type.b, sizeof (type.b));
        written += sizeof(type.b);
      }
    else
      {
        if (dst)
  	{
  	  if (ELFCLASS64 == class)
  	    {
  	      memcpy(dst + written, "\x81\xec", 2);
  	      written ++;
  	    }
  	  else
  	    memcpy(dst + written, "\x2d", 1);
  	}
        written ++;
        if ((uint64_t)type.b4 == type.b8 && type.b4 < 0x80000000)
  	{
  	  if (dst)
  	    memcpy(dst + written, &type.b4, sizeof (type.b4));
  	  written += sizeof(type.b4);
  	}
        else
  	{
  	  if (dst)
  	    memcpy(dst + written, &type.b8, sizeof (type.b8));
  	  written += sizeof(type.b8);
  	}
      }
    return (written);
  }

  size_t ft_sub_asm_rsp_buffer_byte(void *dst, uint64_t offset, unsigned char value, unsigned char class)//mov the dst pointer to not overwrite.
  {
    size_t	written;
    const t_type	off = (t_type){.b8 = offset};

    written = 0;
    if (class == ELFCLASS32 && off.b8 != off.b4)
      return (0);
    if (dst)
      memcpy(dst, "\x80", 1);
    written++;
      /* } */

    if (!off.b8)
      {
        if (dst)
  	memcpy(dst + written, "\x2c", 1);//sub byte
        written++;
      }
    else if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80)
      {
        if (dst)
  	memcpy(dst + written, "\x6c", 1);//sub byte
        written++;
      }
    else 
      {
        if (dst)
  	memcpy(dst + written, "\xac", 1);//sub byte
        written++;
      }
    if (dst)
      memcpy(dst + written, "\x24", 1);// rsp pointer
    written++;
    if (off.b8)
      {
        if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80)
  	{
  	  if (dst)
  	    memcpy(dst + written, &off.b, sizeof(off.b));

  	  written += sizeof (off.b4);
  	}
      }
    if (dst)
      memcpy(dst + written, &value, sizeof (value));
    written += sizeof(value);
    return (written);
  }

  size_t set_xor_register_register(void *dst, t_registers reg, unsigned char class)
  {
    const char bytecode_regmap[] = {'\xc0', '\xc0', '\xdb', '\xdb', '\xc9', '\xc9', '\xd2', '\xd2', '\xff', '\xff', '\xf6', '\xf6', '\xed', '\xed', '\xe4', '\xe4', '\xc0', '\xc9', '\xd2','\xdb','\xe4', '\xed', '\xf6', '\xff'};
    size_t bytes;

    if (reg > R15 || (ELFCLASS32 == class && reg >= R8))
      return (-1);
    bytes = 0;
    if (ELFCLASS64 == class)
      {
        if (dst && reg >= R8 && reg <= R15)
  	memcpy(dst + bytes, "\x4D", 1);
        else if (dst)
  	memcpy(dst + bytes, "\x48", 1);
        bytes++;
      }
    if (dst)
      memcpy(dst + bytes, "\x31", 1);
    bytes++;
    if (dst)
      memcpy(dst + bytes, &bytecode_regmap[reg], 1);
    return (++bytes);
  }

  size_t set_xor_register_size(void *dst, unsigned char class)
  {
    size_t xor_register_size;
    t_registers reg;

    if (ELFCLASS64 == class)
      reg = RAX;// RAX gonna hold the value of entrypoint.
    else
      reg = EAX;
    xor_register_size = 0;
    while ((ELFCLASS64 == class && reg <= R15) || reg < ESP)
      {
        if (reg == R13)
  	reg++;
        if (dst)
  	xor_register_size += set_xor_register_register(dst + xor_register_size, reg, class);
        else
  	xor_register_size += set_xor_register_register(NULL, reg, class);
        if (reg >= R8)
  	reg++; 
        else
  	reg += 2;
        if (reg == RSP)//No xor rsp
  	reg += 2;
      }
    return (xor_register_size);
  }

  size_t add_r13(void *dst, uint64_t value)
  {
    size_t written;
    t_type val = (t_type){.b8=value};

    written = 0;
    if (dst)
      memcpy(dst+ written, "\x49\x81\xc5", 3);
    written +=3;
    if (val.b4 == val.b8)
      {
        if (dst)
  	  memcpy(dst + written, &val.b4, sizeof(val.b4));
        written += sizeof(val.b4);
      }
    else
      {
        if (dst)
  	  memcpy(dst + written, &val.b8, sizeof(val.b8));
        written += sizeof(val.b8);
      }
    return (written);
  }

  size_t set_jmp_entrypoint(void *payload, uint64_t entrypoint, unsigned char class)
  {
    t_elf_addr u_entrypoint = (t_elf_addr){.b64 = entrypoint};
    size_t offset = 0;

    if (class == ELFCLASS64)
      {
        /* memcpy(payload, "\x48", 1); //rex prefix x64 */
        if (payload)
  	memcpy(payload, "\x49", 1);
        offset++;
        if (payload)
  	memcpy(payload + offset, "\xbd", 1); //mov r13
        offset++;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b64, sizeof (u_entrypoint.b64));
        offset += sizeof (u_entrypoint.b64);
        if (payload)
  	{
  	offset += pop_r13(payload + offset);
  	offset += add_r13(payload + offset, u_entrypoint.b64);
  	offset += set_jmp_r13(payload + offset);
  	}
        else
  	{
  	offset += pop_r13(NULL);
  	offset += add_r13(NULL, u_entrypoint.b64);
  	offset += set_jmp_r13(NULL);
  	}
      }
    else
      {
        if (payload)
  	memcpy(payload + offset, "\x81\x04\x24", 3);// add DWORD [esp]
        offset+= 3;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b32, sizeof (u_entrypoint.b32));
        offset += sizeof (u_entrypoint.b32);
        if (payload)
  	memcpy(payload + offset, "\xc3", 1);//ret
        offset++;
      }
    return (offset);
  }

  static t_elf_ph get_n_loadable_and_exec_phdr(t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t n)
  {
    t_elf_ph phdr;
    uint64_t ph_ndx;
    uint64_t times_loadable_and_exec;

    ph_ndx = 0;
    times_loadable_and_exec = 0;
    while (++ph_ndx < phnum)
      {
        phdr = elf_get_program_header(ehdr, ph_ndx);
        if (PT_LOAD == elf_get_program_header_type(&phdr, class) && PF_X & elf_get_program_header_flags(&phdr, class))
  	times_loadable_and_exec++;
        if (times_loadable_and_exec == n)

  static bool increment_offset_phdr(t_elf_h *ehdr, uint64_t offset, uint64_t phnum, unsigned char class)

  uint32_t get_new_e_entrypoint(t_elf_h *ehdr, unsigned char class)
  {
    uint64_t phnum = elf_get_elfh_phnum(ehdr);
    uint64_t shnum = elf_get_elfh_shnum(ehdr);
    uint64_t new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    uint64_t new_phdr_offset = elf_get_highest_offset_avalaible(ehdr, class, shnum);

    const uint64_t new_align = 0x1000;
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    if (increment_offset_phdr(ehdr, new_phdr_offset, phnum, class))// I think shouldn't go throught this way.
      new_phdr_offset += phdr_struct_size;
    if (new_phdr_vaddr % new_align == new_phdr_offset % new_align) 
      printf("all fine with addr and offset aligned\n");
    else
      {
        if (new_phdr_vaddr % new_align > new_phdr_offset % new_align)
  	new_phdr_offset += new_phdr_vaddr % new_align - new_phdr_offset % new_align;
        else
  	new_phdr_vaddr += new_phdr_offset % new_align - new_phdr_vaddr % new_align;
      }
    if (new_phdr_offset % 16)
      {
        new_phdr_vaddr += 16 - new_phdr_vaddr % 16;
        new_phdr_offset += 16 - new_phdr_offset % 16;
      }
    return ((uint32_t)new_phdr_vaddr);
  }

  int load_stage_buffer(int class, char **stage_buffer, size_t *stage_fd_size, char *filename_x86, char *filename_x64)
  {
      int stage_fd;
      
      // Validate input parameters
      if (!stage_buffer || !stage_fd_size)
          return (1);
      
      // Initialize output parameters
      ,*stage_buffer = NULL;
      ,*stage_fd_size = 0;
      
      stage_fd = open(ft_ternary(ELFCLASS32 == class, filename_x86,filename_x64), O_RDONLY);
      if (stage_fd < 0 || get_file_size(stage_fd, (off_t *)stage_fd_size) || !*stage_fd_size)
      {
          if (stage_fd >= 0)
              close(stage_fd);
          return (1);
      }
      
      ,*stage_buffer = malloc(*stage_fd_size);
      if (!(*stage_buffer))
      {
          close(stage_fd);
          return (5);
      }
      
      if (read(stage_fd, *stage_buffer, *stage_fd_size) != (ssize_t) *stage_fd_size)
      {
          free(*stage_buffer);
          ,*stage_buffer = NULL;
          close(stage_fd);
          return (5);
      }
      
      close(stage_fd);
      return (0);
  }

  static uint64_t elf_get_closest_higher_p_offset(const t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t p_offset)
  {
    t_elf_ph phdr;
    uint64_t ph_ndx;
    uint64_t closest_offset = -1;
    uint64_t cur_offset;
    ph_ndx = 0;
    while (ph_ndx < phnum)
      {
        phdr = elf_get_program_header(ehdr, ph_ndx);
        cur_offset = elf_get_program_header_offset(&phdr, class);
        if (cur_offset > p_offset && cur_offset < closest_offset)
  	closest_offset = cur_offset;
        ph_ndx++;
      }
    return (closest_offset);
  }

  static bool elf_is_a_avalaible_gap_in_exec_load_phdr(const t_elf_h *ehdr, unsigned char class, uint64_t phnum, size_t gap_size, uint64_t *ret_vaddr, uint64_t *ret_offset)
  {
    if (!ehdr || !phnum)
      return (false);
    if (!gap_size)
      return (true);
    t_elf_ph phdr;
    uint64_t ph_ndx;
    uint64_t cur_vaddr_top;//cur
    uint64_t cur_offset_top;//cur
    uint64_t top_vaddr;
    uint64_t top_offset;//get closest higher offset.
    uint64_t avalaible_vaddr;
    uint64_t avalaible_offset;

    ph_ndx = 0;
    while (ph_ndx < phnum)
      {
        phdr = elf_get_program_header(ehdr, ph_ndx);
        if (!(PT_LOAD == elf_get_program_header_type(&phdr, class) && PF_X & elf_get_program_header_flags(&phdr, class)))
  	{
  	      ph_ndx++;
  	      continue;
  	}
        cur_vaddr_top = elf_get_program_header_vaddr(&phdr, class) + elf_get_program_header_memsz(&phdr, class);
        top_vaddr = get_alignment_effective_addr(0x1000, cur_vaddr_top);
        avalaible_vaddr = top_vaddr - cur_vaddr_top;
        if (avalaible_vaddr < gap_size)
  	{
  	  ph_ndx++;
  	  continue;
  	}
        cur_offset_top = elf_get_program_header_offset(&phdr, class) + elf_get_program_header_filesz(&phdr, class);
        top_offset = elf_get_closest_higher_p_offset(ehdr, class, phnum, cur_offset_top);
        if (top_offset == (uint64_t) -1)
  	top_offset = elf_get_elfh_shoff(ehdr);
        avalaible_offset = top_offset - cur_offset_top;
        if (avalaible_offset >= gap_size)
  	{
  	  ,*ret_offset = avalaible_offset;
  	  ,*ret_vaddr = cur_vaddr_top - 1;
  	  return (true);
  	}
        ph_ndx++;
      }
    return (false);
  }

  int generate_payload(void *elf_mmap, size_t elf_size, char **payload_ptr, size_t *payload_size, char **stage3, size_t *stage3_size, void **new_elf_buffer)
  {
    const t_elf_h ehdr = (const t_elf_h){.elf32 = elf_mmap};
    const unsigned char 	class = elf_get_elfh_e_ident_class(elf_mmap);
    const uint64_t phnum = elf_get_elfh_phnum(&ehdr);

    off_t stage3_fd_size;
    char *stage3_buffer;
    if (load_stage_buffer(class, &stage3_buffer, (size_t *)&stage3_fd_size, "bytecode/x86/stage3.bin", "bytecode/x64/stage3.bin"))
        return (5);
    uint64_t avalaible_vaddr;
    uint64_t avalaible_offset;
    if (elf_is_a_avalaible_gap_in_exec_load_phdr((void *)&ehdr, class, phnum, stage3_fd_size, &avalaible_vaddr, &avalaible_offset))
      {
        ,*new_elf_buffer = malloc(elf_size);
        if (!*new_elf_buffer)
  	{
  	  free(stage3_buffer);
  	  return (5);
  	}
        memcpy(*new_elf_buffer, elf_mmap, elf_size);
        memcpy(*new_elf_buffer + avalaible_offset, stage3_buffer, stage3_fd_size);
        free(stage3_buffer);
        const uint64_t ndx_ph = elf_get_program_ndx_from_addr((void *)&ehdr, class, phnum, avalaible_vaddr);
        if (ndx_ph >= phnum)
  	  return (5);
        uint64_t ehdr_struct_size = elf_get_struct_size_elf_header(elf_mmap);
        t_elf_ph phdr =(t_elf_ph){.elf32 = *new_elf_buffer + (elf_get_struct_size_elf_ph(class) * ndx_ph) + ehdr_struct_size};

        elf_phdr_set_filesz(&phdr, elf_get_program_header_filesz(&phdr, class) + stage3_fd_size, class);
        elf_phdr_set_memsz(&phdr, elf_get_program_header_memsz(&phdr, class) + stage3_fd_size, class);
      }
    else
      {
        ,*stage3 = stage3_buffer;
        ,*stage3_size = stage3_fd_size;
      }
    stage3_buffer = NULL;
    
    char *stage2_buffer;
    size_t stage2_fd_size;
    if (load_stage_buffer(class, &stage2_buffer, (size_t *)&stage2_fd_size, "bytecode/x86/stage2.bin", "bytecode/x64/stage2.bin"))
      return (5);
    // Modify need data
    size_t offset_mod_stage2[] = {0x8};
    uint32_t tmp_val_stage2[] = {elf_size};
    uint64_t i = 0;
    while (i < sizeof(offset_mod_stage2) / sizeof(*offset_mod_stage2))
      {
        memcpy(stage2_buffer + offset_mod_stage2[i], &tmp_val_stage2[i], sizeof(*tmp_val_stage2));
        i++;
      }

    // Join with elf file
    size_t stage2_elf_buffer_size = stage2_fd_size + elf_size;
    char *stage2_elf_buffer = malloc(stage2_elf_buffer_size);
    if (!stage2_elf_buffer)
      {
        free(stage2_buffer);
        return (5);
      }
    memcpy(stage2_elf_buffer, stage2_buffer, stage2_fd_size);
    free(stage2_buffer);
    memcpy(stage2_elf_buffer + stage2_fd_size, elf_mmap, elf_size);
    // Compress the data.
    t_list *lz77_buffer = lz77_compress((unsigned char *)stage2_elf_buffer, stage2_elf_buffer_size);
    free(stage2_elf_buffer);
    if (!lz77_buffer)
        return (5);
    // Get the array of the compress lst
    size_t stage2_elf_compress_size;
    unsigned char *stage2_elf_compress = lz77_get_compress_arr(lz77_buffer, &stage2_elf_compress_size);
    ft_lstclear(&lz77_buffer, free);
    if (!stage2_elf_compress)
        return (5);
    // Add the original size as header
    size_t stage2_elf_compress_header_size = stage2_elf_compress_size + sizeof(size_t);
    char *stage2_elf_compress_header = malloc(stage2_elf_compress_header_size);
    if (!stage2_elf_compress_header)
      {
        free(stage2_elf_compress);
        return (5);
      }
    memcpy(stage2_elf_compress_header, &stage2_elf_compress_size, sizeof(size_t));
    memcpy(stage2_elf_compress_header + sizeof(size_t) , stage2_elf_compress, stage2_elf_compress_size);
    free(stage2_elf_compress);

  static int write_program_headers(t_elf_h *ehdr, int packed, uint64_t phnum, uint64_t new_phdr_vaddr,
                                   unsigned char class, uint64_t payload_size, uint64_t *writed/* , uint64_t *stage3_size */)
  {
    const uint64_t ehdr_struct_size = elf_get_struct_size_elf_header((char *)ehdr->elf32);
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    t_elf_ph phdr;
    uint64_t ndx_ph;
    uint64_t p_offset;
    uint64_t p_filesz;
    uint64_t p_memsz;
    uint64_t p_align;
    uint64_t p_vaddr;
    uint64_t p_paddr;
    uint64_t p_type;
    uint64_t p_flags;
    t_elf_ph phdr_cp = {.elf32 = malloc(phdr_struct_size)};

    if (!phdr_cp.elf32)
      return (1);
    ndx_ph = 0;
    phdr = elf_get_program_header(ehdr, ndx_ph);

    p_align = 0x1000;
    p_filesz = p_align;
    p_memsz = elf_get_highest_used_vaddr_avalaible(ehdr, class, phnum);
    p_offset = 0;
    p_vaddr = elf_get_program_header_vaddr(&phdr, class);
    p_memsz -= p_vaddr;
    p_paddr = elf_get_program_header_paddr(&phdr, class);
    p_type = PT_LOAD;
    p_flags = PF_R | PF_W;
    elf_phdr_set_filesz(&phdr_cp, p_filesz, class);
    elf_phdr_set_memsz(&phdr_cp, p_memsz, class);
    elf_phdr_set_offset(&phdr_cp, p_offset, class);
    elf_phdr_set_vaddr(&phdr_cp, p_vaddr, class);
    elf_phdr_set_paddr(&phdr_cp, p_paddr, class);
    elf_phdr_set_type(&phdr_cp, p_type, class);
    elf_phdr_set_flags(&phdr_cp, p_flags, class);
    elf_phdr_set_align(&phdr_cp, p_align, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      {
        free(phdr_cp.elf32);
        return (2);
      }
    ,*writed += phdr_struct_size;//first header
    p_flags = PF_R | PF_X;
    p_vaddr = new_phdr_vaddr;
    p_paddr = p_vaddr;
    p_memsz = payload_size + ehdr_struct_size + phdr_struct_size * (2 + elf_is_gnu_stack_phdr(ehdr, class, phnum));

    p_filesz = p_memsz;
    elf_phdr_set_filesz(&phdr_cp, p_filesz, class);
    elf_phdr_set_memsz(&phdr_cp, p_memsz, class);
    elf_phdr_set_vaddr(&phdr_cp, p_vaddr, class);
    elf_phdr_set_paddr(&phdr_cp, p_paddr, class);
    elf_phdr_set_flags(&phdr_cp, p_flags, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      {
        free(phdr_cp.elf32);
        return (3);
      }
    ,*writed += phdr_struct_size;//second header
    free(phdr_cp.elf32);
    if (elf_is_gnu_stack_phdr(ehdr, class, phnum))
      {
        ndx_ph = elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum);
        phdr = elf_get_program_header(ehdr, ndx_ph);
        if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
  	  return (4);
        ,*writed += phdr_struct_size;
      }
    return (0);
  }

      jb     .copy_literal      ; Jump if distance > pos

  // Generic function to calculate memory range for ELF segments
  void calculate_elf_memory_range(Elf64_Phdr *phdr, int phnum, uint64_t *min_vaddr, uint64_t *max_vaddr)
  {
      ,*min_vaddr = -1;
      ,*max_vaddr = 0;
      
      while (phnum--) {
          if (phdr[phnum].p_type == PT_LOAD) {
              if (phdr[phnum].p_vaddr < *min_vaddr)

      uint64_t page_vaddr = phdr->p_vaddr & ~(PAGE_SIZE - 1);
      uint64_t map_end = (phdr->p_offset + phdr->p_memsz + PAGE_SIZE - 1) & ~(PAGE_SIZE - 1);
      size_t map_size = map_end - page_offset;
      
      // Map at the correct address with MAP_FIXED
      int prot = PROT_READ;
      if (phdr->p_flags & PF_W || fd == -1) prot |= PROT_WRITE;
      if (phdr->p_flags & PF_X && fd != -1) prot |= PROT_EXEC;
      int flags = MAP_FIXED | MAP_PRIVATE;
      if (fd == -1)
        flags |= MAP_ANONYMOUS;
      void *map_addr = (char *)base + (page_vaddr - min_vaddr);
      void *result = mmap(map_addr, map_size, prot, 
  			flags, fd, page_offset);
      if (result == MAP_FAILED)
        return -1;
      if (fd == -1) {
          // Copy data from ELF buffer to mapped memory
          memcpy((char *)result + (phdr->p_vaddr - page_vaddr),
                 (char *)elf_ptr + phdr->p_offset,
                 phdr->p_filesz);
  	prot = PROT_READ;
  	if (phdr->p_flags & PF_W) prot |= PROT_WRITE;
  	if (phdr->p_flags & PF_X) prot |= PROT_EXEC;
  	if (mprotect(map_addr, map_size, prot) < 0)
  	  return -1;
      }
      return 0;
  }

  // Generic ELF unpacking function that works for both main ELF and interpreter
  uint64_t load_elf_generic(void *ptr, uint64_t aslr_offset, uint64_t target_vaddr, int fd)
  {
      Elf64_Ehdr *elf = (Elf64_Ehdr *)ptr;
      Elf64_Phdr *elf_ph = (Elf64_Phdr *)((char *)ptr + elf->e_phoff);
      
      uint64_t min_vaddr, max_vaddr;
      
      // Calculate memory range needed
      calculate_elf_memory_range(elf_ph, elf->e_phnum, &min_vaddr, &max_vaddr);
      size_t total_size = max_vaddr - min_vaddr;
      
      // Reserve entire address space with PROT_NONE
      void *base_addr;
      if (target_vaddr)
        target_vaddr += aslr_offset;
      base_addr = mmap((void *)target_vaddr, total_size, 
                          PROT_NONE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
      
      if (base_addr == MAP_FAILED)
          return (uint64_t)-1;
      
      // Load each segment
      for (int i = 0; i < elf->e_phnum; i++) {
          if (load_elf_segment(base_addr, ptr, &elf_ph[i], min_vaddr, fd) < 0) {
              munmap(base_addr, total_size);
              return (uint64_t)-1;
          }
      }
      
      uint64_t entrypoint = (uint64_t)base_addr + (elf->e_entry - min_vaddr);
      return entrypoint;
  }

  // rdi base.
  uint64_t get_at_interp_x64(void *ptr)
  {
      Elf64_Ehdr *elf = (Elf64_Ehdr *)ptr;
      Elf64_Phdr *elf_ph = (Elf64_Phdr *)((char *)ptr + elf->e_phoff);
      uint64_t ndx_ph = 0;
      char *interp_str;
      int interp_fd;
      char interp_elfh[1024];
      
      // Find PT_INTERP segment
      while (ndx_ph < elf->e_phnum && elf_ph[ndx_ph].p_type != PT_INTERP)
          ndx_ph++;
      
      if (ndx_ph == elf->e_phnum)
          return 0;  // No interpreter found
      
      interp_str = (char *)ptr + elf_ph[ndx_ph].p_offset;
      interp_fd = open(interp_str, O_RDONLY);
      if (interp_fd < 0)
          return (uint64_t)-1;
      
      if (read(interp_fd, &interp_elfh[0], 1024) < 1024) {
          close(interp_fd);
          return (uint64_t)-1;
      }
      
      // Use the generic unpacker for the interpreter
      uint64_t result = load_elf_generic(&interp_elfh[0], 0, 0, interp_fd);
      
      close(interp_fd);
      return result;
  }

  size_t lz77_decompress_asm(const unsigned char *in, size_t in_size, unsigned char *out) {
    size_t orig_size;
    size_t pos = 0;
    uint16_t dl;
    size_t len, dist;
    const unsigned char *p = in + sizeof(orig_size);

    // Validate input
    if (!in || in_size < sizeof(orig_size) || !out)
      goto error;

    // Read original size
    orig_size = *((size_t *) in);
    if (!orig_size)
      goto error;

    // Decompress loop
    while ((p + 3 <= in + in_size) && pos < orig_size) {
      // Read distance/length as little-endian
      dl = p[0] | (p[1] << 8);
      dist = (dl >> 4) & 0xFFF;
      len = (dl & 0xF);
      if (len) len += 2;

      // Copy match data
      if (len && dist && dist <= pos) {
        if (len > orig_size - pos) 
  	len = orig_size - pos;
    	  
        unsigned char *dst = out + pos;
        unsigned char *src = out + (pos - dist);
    	  
        while (len--) 
  	,*dst++ = *src++;
        pos += (dst - (out + pos));  // Update pos
      }
      // Copy literal byte
      if (pos < orig_size)
        out[pos++] = p[2];
      p += 3;
    }
    // Check if we decompressed all data
    if (pos != orig_size)
      goto error;
    return 0;

   error:
    return -1;
  }
