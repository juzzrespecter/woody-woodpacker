    t_list	*ft_lstnew(void *content)
    {
    	t_list	*node;

    	node = malloc(sizeof(t_list));
    	if (!node)
    		return (NULL);
    	node->content = content;
    	node->next = NULL;
    	return (node);
    }

    t_lz77 *create_lz77_tkn(size_t len, size_t distance, char byte)
    {
      t_lz77 *tkn = malloc(sizeof(*tkn));

      if (tkn)
        {
          if (len >= COMPRESS_LZ77_MIN_MATCH_LEN)
    	tkn->len = len - (COMPRESS_LZ77_MIN_MATCH_LEN - 1);
          else
    	tkn->len = 0;
          /* tkn->len = len; */
          tkn->distance = distance;
          tkn->byte = byte;
        }
      return (tkn);
    }

    t_list	*ft_lstlast(t_list *lst)
    {
    	if (!lst)
    		return (NULL);
    	while (lst->next)
    		lst = lst->next;
    	return (lst);
    }

    int	ft_lstsize(t_list *lst)
    {
    	int	lst_num;

    	if (!(size_t)lst)
    		return (0);
    	lst_num = 1;
    	while (lst->next)
    	{
    		lst_num++;
    		lst = lst->next;
    	}
    	return (lst_num);
    }

    void	ft_lstadd_back(t_list **lst, t_list *new)
    {
    	t_list	*lst_last;

    	if (!new || !lst)
    		return ;
    	if (*lst == NULL)
    		,*lst = new;
    	else
    	{
    		lst_last = ft_lstlast(*lst);
    		lst_last->next = new;
    	}
    }

    void	ft_lstdelone(t_list *lst, void (*del)(void *))
    {
    	if (!lst || !del)
    		return ;
    	del(lst->content);
    	free(lst);
    }

    void	ft_lstclear(t_list **lst, void (*del)(void *))
    {
    	t_list	*node;
    	t_list	*tmp;

    	if (!lst || !*lst)
    		return ;
    	node = *lst;
    	while (node)
    	{
    		tmp = node->next;
    		ft_lstdelone(node, del);
    		node = tmp;
    	}
    	,*lst = NULL;
    }

    void add_sliding_window(char *buffer, size_t *buffer_start, size_t *buffer_end, size_t *buffer_size, const char *data, size_t len)
    {
        size_t i;
        
        // Add data to ring buffer - new data goes at the end
        i = 0;
        while (i < len)
        {
            // Add new byte to end of buffer
            buffer[*buffer_end] = data[i];
            ,*buffer_end = (*buffer_end + 1) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
            
            // Update size
            if (*buffer_size < COMPRESS_LZ77_SLIDING_WINDOW_SIZE)
                (*buffer_size)++;
            else
                ,*buffer_start = (*buffer_start + 1) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
    	i++;
        }
        if (*buffer_size == COMPRESS_LZ77_SLIDING_WINDOW_SIZE && *buffer_start == *buffer_end)
          ,*buffer_start = (*buffer_start + 1) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;

    }

    // Access buffer from end backwards (most recent first)
    char get_sliding_window_char_from_end(const char *buffer, size_t buffer_end, size_t buffer_size, size_t backward_index)
    {
        if (!buffer_size || backward_index >= buffer_size)
            return 0; // Out of bounds
       size_t target_pos = (buffer_end - 1 - backward_index + COMPRESS_LZ77_SLIDING_WINDOW_SIZE) 
                           % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
        return buffer[target_pos]; 
    }

    char get_sliding_window_char(const char *buffer, size_t buffer_start, size_t buffer_size, size_t index)
    {
      if (!buffer || !buffer_size || index >= buffer_size)
        return (0);
      size_t actual_index = (buffer_start + index) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
      return (buffer[actual_index]);
    }
    int lz77_compress_find_best_match(const char *sliding_window, size_t sliding_window_start, size_t sliding_window_end, size_t sliding_window_buffer_size, const char *src, size_t src_len, size_t *best_distance, size_t *best_len)
    {
      size_t search_pos;
      size_t match_len;
      size_t window_ndx;
      size_t sliding_window_start_reached;

      if (!sliding_window||! src)
        return (-1);
      ,*best_distance = 0;
      ,*best_len = 0;
      sliding_window_start_reached = 0;
      search_pos = sliding_window_end;//
      match_len = 0;
      while (sliding_window_buffer_size && !sliding_window_start_reached && src_len && *best_len != COMPRESS_LZ77_MAX_MATCH_LEN)
        {
          search_pos = (search_pos - 1 + COMPRESS_LZ77_SLIDING_WINDOW_SIZE) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
          if (search_pos == sliding_window_start)
    	sliding_window_start_reached = 1;
          match_len = 0;
          window_ndx = (search_pos + match_len) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
          while (match_len < COMPRESS_LZ77_MAX_MATCH_LEN && match_len < src_len && window_ndx < sliding_window_buffer_size
    	     && get_sliding_window_char(sliding_window, sliding_window_start, sliding_window_buffer_size, window_ndx) == src[match_len])
    	{
    	  match_len++;
    	  window_ndx = (search_pos + match_len) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
    	}
          if (COMPRESS_LZ77_MAX_MATCH_LEN < match_len)
    	match_len = COMPRESS_LZ77_MAX_MATCH_LEN;
          if (match_len >= COMPRESS_LZ77_MIN_MATCH_LEN &&match_len > *best_len)
    	{
    	  ,*best_len = match_len;
    	  ,*best_distance = (sliding_window_end - search_pos + COMPRESS_LZ77_SLIDING_WINDOW_SIZE) % COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
    	  if (!*best_distance || *best_distance > sliding_window_buffer_size)
    	    {
    	      ,*best_distance = 0;
    	      ,*best_len = 0;
    	    }
    	}
        }
      return (0);
    }

    t_list *lz77_compress_list(const char *src, size_t src_len)
    {
      char sliding_window[COMPRESS_LZ77_SLIDING_WINDOW_SIZE];
      size_t sliding_window_end = 0;
      size_t sliding_window_start = 0;
      size_t sliding_window_buffer_size = 0;
      size_t distance;
      size_t len;
      t_list *tokens_lst;
      t_list *token_lst;
      t_lz77 *token;
      char		byte;
      size_t token_count;
      size_t total_input_processed;
      size_t og_src_len = src_len;
      size_t literal_count = 0;
      size_t match_count = 0;
      size_t total_match_length = 0;

      token_count = 0;
      total_input_processed = 0;
      tokens_lst = NULL;
      while (src_len)
        {
          if (lz77_compress_find_best_match(&sliding_window[0], sliding_window_start, sliding_window_end, sliding_window_buffer_size, src, src_len, &distance, &len))
    	{
    	  ft_lstclear(&tokens_lst, free);
    	  return (tokens_lst);
    	}
          byte = 0;
          if (len >= src_len)
    	printf("Ignored next byte inside the token\n");
          else
    	byte = src[len];

          token = create_lz77_tkn(len, distance, byte);
          if (byte != '\n')
    	printf("(%lu, %lu, %c)\t", distance, len, byte);
          else
    	printf("(%lu, %lu, \\n)\t", distance, len);
          if (!len)
    	literal_count++;
          else
    	{
    	  match_count++;
    	  total_match_length += len;
    	}
    	
          if (!token)
    	{
    	  ft_lstclear(&tokens_lst, free);
    	  return (tokens_lst);
    	}
          token_lst = ft_lstnew(token);
          if (!token_lst)
    	{
    	  free(token);
    	  ft_lstclear(&tokens_lst, free);
    	  return (tokens_lst);
    	}
          ft_lstadd_back(&tokens_lst, token_lst);
          len++;// INFO If delete this line all len should be + 1
          add_sliding_window(sliding_window, &sliding_window_start, &sliding_window_end, &sliding_window_buffer_size, src, len);
          src += len;
          if (len > src_len)
    	{
    	  len = src_len;
    	  src_len = 0;
    	}
          else
    	src_len -= len;
          token_count++;
          total_input_processed += len;
          printf("Total tokens: %zu\t", token_count);
          printf("Total input processed: %zu\t", total_input_processed);
          printf("Original input size: %zu\t", og_src_len);
          printf("window_start %lu\t", sliding_window_start);
          printf("window_end %lu\t", sliding_window_end);
          printf("Tokens per byte: %.2f\n", (double)token_count / total_input_processed);
        }
      printf("Literals: %ld, Matches: %ld\n", literal_count, match_count);
      printf("Average match length: %.2f\n", (double)total_match_length / match_count);
      printf("Compression efficiency: %.2f%%\n", (double)(literal_count + match_count) / total_input_processed * 100);
      return (tokens_lst);
    }

    int lz77_compress(char *src, size_t src_len, int fd)
    {
      t_list *tokens = lz77_compress_list(src, src_len);
      t_list *tmp;
      t_lz77 *token;
      size_t compress_size;

      if (!tokens)
        return (-1);
      tmp = tokens;
      // Write original size.
      compress_size = ft_lstsize(tmp) * sizeof(*token) + sizeof(compress_size);
      printf("compress_size %lu, src_size %lu, compress_ratio %f\n", compress_size, src_len, (double)compress_size / src_len);
      if (compress_size > src_len)
        printf("don't compress\n");
      compress_size = src_len;
      write(fd, &compress_size, sizeof(compress_size));
      while (tmp)
        {
          token = tmp->content;
          if (!token)//Shouldn't happend.
    	return (-1);
          /* if (tmp == tokens) */
          write(fd, token, sizeof(*token));
          tmp = tmp->next;
        }
      ft_lstclear(&tokens, free);
      return (0);
    }


    int lz77_decompress(const char *compressed_data, size_t compressed_size, int fd)
    {w
      size_t original_size;
      char *output;
      char window[COMPRESS_LZ77_SLIDING_WINDOW_SIZE] = {0};
      size_t window_start = 0, window_end = 0, window_size = 0, output_pos = 0;
      const char *current = compressed_data + sizeof(size_t);
      size_t length;
      t_lz77 *token;
      if (!compressed_data || compressed_size < sizeof(size_t))
        return (-1);

      memcpy(&original_size, compressed_data, sizeof(size_t));
      output = malloc(original_size); 
      if (!output)
        return (-1);
        
      // Use your existing sliding window functions
      while ((current + sizeof(*token)) <= (compressed_data + compressed_size) && output_pos < original_size) {
        token = (t_lz77*)current;
            
        // Unified processing
        length = 0;
        if (token->len > 0)
          length = (token->len + 2);
            
        // Copy referenced data using your get function
        for (size_t i = 0; token->distance && i < length && output_pos < original_size; i++) {
          // For decompression, we want to read from (distance) positions back
    	size_t backward_idx = window_end - length; // 1 = most recent
    	printf("backward_idx %lu\n, window_size %lu\n", backward_idx, window_size);
    	if (backward_idx == window_size)
    	    break;
    	char byte = get_sliding_window_char(window, window_start, window_size, backward_idx + i);
    	output[output_pos++] = byte;
          }
            
        // Add next character
        if (output_pos < original_size)
          output[output_pos++] = token->byte;
            
        // Update window using your add function
        add_sliding_window(window, &window_start, &window_end, &window_size,
    		       output + (output_pos - length - 1), length + 1);
            
        current += sizeof(t_lz77);
      }
        
      if (output_pos == original_size) {
        write(fd, output, output_pos);
        free(output);
        return (0);
      }
        
      free(output);
      return (-1);
    }

        phdr = elf_get_program_header(ehdr, i);

  size_t ft_je(unsigned char *dst, size_t distance)
  {

  size_t ft_mov_asm_rsp(void *dst, size_t offset, size_t value) //mov the dst pointer to not overwrite.
  {
    size_t	written;
    const t_type	type = (t_type){.b8 = value};
    const t_type	off = (t_type){.b8 = offset};

    written = 0;
    if (dst)
      memcpy(dst, "\xc6", 1);
    written++;
    if (!off.b8)
      {
        if (dst)
          memcpy(dst + written, "\x04", 1);
      }
    else if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80) // if offset and byte syze
      {
        if (dst)
          memcpy(dst + written, "\x44", 1);
      }
    else // if offset and > byte size
      {
        if (dst)
          memcpy(dst + written, "\x84", 1);
      }
    written++;
    if (dst)
      memcpy(dst + written, "\x24", 1);
    written++;
    if (off.b8)
      {
        /* if (dst) */
        /* printf("off -> b value %u, b4 value %u, b8 value %lu\n", (uint32_t)off.b, off.b4, off.b8); */
        if ((uint32_t)off.b  == off.b4 && (uint64_t)off.b4 == off.b8 && off.b < 0x80)
          {
            if (dst)
              memcpy(dst + written, &off.b, sizeof (off.b)); 
            written += sizeof (off.b);
          }
        else if ((uint64_t)off.b4 == off.b8 && off.b4 < 0x80000000)
          {
            if (dst)
              memcpy(dst + written, &off.b4, sizeof (off.b4)); 
            written += sizeof (off.b4);
          }
        else
          {
            if (dst)
              memcpy(dst + written, &off.b8, sizeof (off.b8)); 
            written += sizeof (off.b8);
          }
      }
    if ((uint32_t)type.b  == type.b4 && (uint64_t)type.b4 == type.b8)
      {
        if (dst)
          memcpy(dst + written, &type.b, sizeof (type.b));
        written += sizeof (type.b);
      }
    else if ((uint64_t)type.b4 == type.b8 && type.b4)
      {
        if (dst)
          memcpy(dst + written, &type.b4, sizeof (type.b4));
        written += sizeof (type.b4);
      }
    else
      {
        if (dst)
          memcpy(dst + written, &type.b8, sizeof (type.b8));
        written += sizeof (type.b8);
      }
    return (written);
  }

  {
    size_t	written;
    const t_type	type = (t_type){.b8 = value};

    if (ELFCLASS32 == class && type.b8 != type.b4)
      return (-1);
    written = 0;
    if (class == ELFCLASS64)
      {
        if (dst)
  	memcpy(dst, "\x48", 1);
        written++;
      }
    if ((uint32_t)type.b  == type.b4 && (uint64_t)type.b4 == type.b8 && type.b < 0x80)
      {
        if (dst)
  	memcpy(dst + written, "\x83", 1);
        written++;
        if (dst)
  	{
  	  /* if (ELFCLASS64 == class) */
  	    memcpy(dst+written, "\xec", 1);
  	  /* else */
  	    /* memcpy(dst+written, "\xe8", 1); */
  	}
        written++;
        if (dst)
  	memcpy(dst + written, &type.b, sizeof (type.b));
        written += sizeof(type.b);
      }
    else
      {
        if (dst)
  	{
  	  if (ELFCLASS64 == class)
  	    {
  	      memcpy(dst + written, "\x81\xec", 2);
  	      written ++;
  	    }
  	  else
  	    memcpy(dst + written, "\x2d", 1);
  	}
        written ++;
        if ((uint64_t)type.b4 == type.b8 && type.b4 < 0x80000000)
  	{
  	  if (dst)
  	    memcpy(dst + written, &type.b4, sizeof (type.b4));
  	  written += sizeof(type.b4);
  	}
        else
  	{
  	  if (dst)
  	    memcpy(dst + written, &type.b8, sizeof (type.b8));
  	  written += sizeof(type.b8);
  	}
      }
    return (written);
  }

  size_t set_xor_register_register(void *dst, t_registers reg, unsigned char class)
  {
    const char bytecode_regmap[] = {'\xc0', '\xc0', '\xdb', '\xdb', '\xc9', '\xc9', '\xd2', '\xd2', '\xff', '\xff', '\xf6', '\xf6', '\xed', '\xed', '\xe4', '\xe4', '\xc0', '\xc9', '\xd2','\xdb','\xe4', '\xed', '\xf6', '\xff'};
    size_t bytes;

    if (reg > R15 || (ELFCLASS32 == class && reg >= R8))
      return (-1);
    bytes = 0;
    if (ELFCLASS64 == class)
      {
        if (dst && reg >= R8 && reg <= R15)
  	memcpy(dst + bytes, "\x4D", 1);
        else if (dst)
  	memcpy(dst + bytes, "\x48", 1);
        bytes++;
      }
    if (dst)
      memcpy(dst + bytes, "\x31", 1);
    bytes++;
    if (dst)
      memcpy(dst + bytes, &bytecode_regmap[reg], 1);
    return (++bytes);
  }

  size_t set_xor_register_size(void *dst, unsigned char class)
  {
    size_t xor_register_size;
    t_registers reg;

    if (ELFCLASS64 == class)
      reg = RAX;// RAX gonna hold the value of entrypoint.
    else
      reg = EAX;
    xor_register_size = 0;
    while ((ELFCLASS64 == class && reg <= R15) || reg < ESP)
      {
        if (reg == R13)
  	reg++;
        if (dst)
  	xor_register_size += set_xor_register_register(dst + xor_register_size, reg, class);
        else
  	xor_register_size += set_xor_register_register(NULL, reg, class);
        if (reg >= R8)
  	reg++; 
        else
  	reg += 2;
        if (reg == RSP)//No xor rsp
  	reg += 2;
      }
    return (xor_register_size);
  }

  size_t set_jmp_entrypoint(void *payload, uint64_t entrypoint, unsigned char class)
  {
    t_elf_addr u_entrypoint = (t_elf_addr){.b64 = entrypoint};
    size_t offset = 0;

    if (class == ELFCLASS64)
      {
        /* memcpy(payload, "\x48", 1); //rex prefix x64 */
        if (payload)
  	memcpy(payload, "\x49", 1);
        offset++;
        if (payload)
  	memcpy(payload + offset, "\xbd", 1); //mov r13
        offset++;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b64, sizeof (u_entrypoint.b64));
        offset += sizeof (u_entrypoint.b64);
        if (payload)
  	{
  	offset += pop_r13(payload + offset);
  	offset += add_r13(payload + offset, u_entrypoint.b64);
  	offset += set_jmp_r13(payload + offset);
  	}
        else
  	{
  	offset += pop_r13(NULL);
  	offset += add_r13(NULL, u_entrypoint.b64);
  	offset += set_jmp_r13(NULL);
  	}
      }
    else
      {
        if (payload)
  	memcpy(payload + offset, "\x81\x04\x24", 3);// add DWORD [esp]
        offset+= 3;
        if (payload)
  	memcpy(payload + offset, &u_entrypoint.b32, sizeof (u_entrypoint.b32));
        offset += sizeof (u_entrypoint.b32);
        if (payload)
  	memcpy(payload + offset, "\xc3", 1);//ret
        offset++;
      }
    return (offset);
  }

  static size_t copy_payload_to_mmap_aux(unsigned char *dst, size_t offset)
  {
    size_t written;
    written = ft_mov_cl_from_rsp(dst, offset);
    if (dst)
      return (written + ft_mov_rax_from_cl(dst + written, offset));
    return (written + ft_mov_rax_from_cl(dst, offset));
  }

  size_t copy_payload_to_mmap(unsigned char *dst, size_t payload_size)
  {
    size_t written = 0;
    size_t i = 0;

    while (i < payload_size)
      { /* dst + written segfault if dst == NULL*/
        if (dst)
          written += copy_payload_to_mmap_aux(dst + written, i);
        else
          written += copy_payload_to_mmap_aux(dst, i);
        i++;
      }
    return (written);
  }

  static t_elf_ph get_n_loadable_and_exec_phdr(t_elf_h *ehdr, unsigned char class, uint64_t phnum, uint64_t n)
  {
    t_elf_ph phdr;
    uint64_t ph_ndx;
    uint64_t times_loadable_and_exec;

    ph_ndx = 0;
    times_loadable_and_exec = 0;
    while (++ph_ndx < phnum)
      {
        phdr = elf_get_program_header(ehdr, ph_ndx);
        if (PT_LOAD == elf_get_program_header_type(&phdr, class) && PF_X & elf_get_program_header_flags(&phdr, class))
  	times_loadable_and_exec++;
        if (times_loadable_and_exec == n)
  	return (phdr);
      }
    phdr.elf32 = NULL;
    return (phdr);
  }

  static bool increment_offset_phdr(t_elf_h *ehdr, uint64_t offset, uint64_t phnum, unsigned char class)
  {
    uint64_t ehdr_struct_size = elf_get_struct_size_elf_header((char *)ehdr->elf32);
    t_elf_ph phdr_loadable_and_exec_first = get_n_loadable_and_exec_phdr(ehdr, class, phnum, 1);
    uint64_t phdr_loadable_and_exec_first_offset = elf_get_program_header_offset(&phdr_loadable_and_exec_first, class);

    return (offset > ehdr_struct_size && offset < phdr_loadable_and_exec_first_offset);
  }

  uint32_t get_new_e_entrypoint(t_elf_h *ehdr, unsigned char class)
  {
    uint64_t phnum = elf_get_elfh_phnum(ehdr);
    uint64_t shnum = elf_get_elfh_shnum(ehdr);
    uint64_t new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    uint64_t new_phdr_offset = elf_get_highest_offset_avalaible(ehdr, class, shnum);

    const uint64_t new_align = 0x1000;
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    new_phdr_vaddr = elf_get_highest_loadable_vaddr_avalaible(ehdr, class, phnum);
    if (increment_offset_phdr(ehdr, new_phdr_offset, phnum, class))// I think shouldn't go throught this way.
      new_phdr_offset += phdr_struct_size;
    if (new_phdr_vaddr % new_align == new_phdr_offset % new_align) 
      printf("all fine with addr and offset aligned\n");
    else
      {
        if (new_phdr_vaddr % new_align > new_phdr_offset % new_align)
  	new_phdr_offset += new_phdr_vaddr % new_align - new_phdr_offset % new_align;
        else
  	new_phdr_vaddr += new_phdr_offset % new_align - new_phdr_vaddr % new_align;
      }
    if (new_phdr_offset % 16)
      {
        new_phdr_vaddr += 16 - new_phdr_vaddr % 16;
        new_phdr_offset += 16 - new_phdr_offset % 16;
      }
    return ((uint32_t)new_phdr_vaddr);
  }

      /* printf("error on write the generated bytecode\n"); */

    ,*payload_ptr = generated_bytecode;
    return (0);
  }

  {
    bool i = true;
    size_t times = 0;

    while (i && times < num)
      {
        if ((ssize_t)1 != write(packed, "\0", 1))
  	i = false;
        times++;
      }
    return (!i);
  }

  static int write_program_headers(t_elf_h *ehdr, int packed, uint64_t phnum, uint64_t new_phdr_vaddr,
                                   unsigned char class, uint64_t payload_size, uint64_t *writed)
  {
    const uint64_t phdr_struct_size = elf_get_struct_size_elf_ph(class);
    t_elf_ph phdr;
    uint64_t ndx_ph;
    uint64_t p_offset;
    uint64_t p_filesz;
    uint64_t p_memsz;
    uint64_t p_align;
    uint64_t p_vaddr;
    uint64_t p_paddr;
    uint64_t p_type;
    uint64_t p_flags;
    t_elf_ph phdr_cp = {.elf32 = malloc(phdr_struct_size)};

    if (!phdr_cp.elf32)
      return (1);
    ndx_ph = 0;
    phdr = elf_get_program_header(ehdr, ndx_ph);

    p_align = 0x1000;
    p_filesz = p_align;
    p_memsz = elf_get_highest_used_vaddr_avalaible(ehdr, class, phnum);
    p_offset = 0;
    p_vaddr = elf_get_program_header_vaddr(&phdr, class);
    p_paddr = elf_get_program_header_paddr(&phdr, class);
    p_type = PT_LOAD;
    p_flags = PF_R | PF_W;
    elf_phdr_set_filesz(&phdr_cp, p_filesz, class);
    elf_phdr_set_memsz(&phdr_cp, p_memsz, class);
    elf_phdr_set_offset(&phdr_cp, p_offset, class);
    elf_phdr_set_vaddr(&phdr_cp, p_vaddr, class);
    elf_phdr_set_paddr(&phdr_cp, p_paddr, class);
    elf_phdr_set_type(&phdr_cp, p_type, class);
    elf_phdr_set_flags(&phdr_cp, p_flags, class);
    elf_phdr_set_align(&phdr_cp, p_align, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      {
        free(phdr_cp.elf32);
        return (2);
      }
    ,*writed += phdr_struct_size;//first header
    p_flags = PF_R | PF_X;
    p_vaddr = new_phdr_vaddr;
    p_paddr = p_vaddr;
    p_memsz = payload_size;
    p_filesz = payload_size;
    elf_phdr_set_filesz(&phdr_cp, p_filesz, class);
    elf_phdr_set_memsz(&phdr_cp, p_memsz, class);
    elf_phdr_set_vaddr(&phdr_cp, p_vaddr, class);
    elf_phdr_set_paddr(&phdr_cp, p_paddr, class);
    elf_phdr_set_flags(&phdr_cp, p_flags, class);
    if (write(packed, phdr_cp.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
      {
        free(phdr_cp.elf32);
        return (3);
      }
    ,*writed += phdr_struct_size;//second header
    free(phdr_cp.elf32);
    if (elf_is_gnu_stack_phdr(ehdr, class, phnum))
      {
        ndx_ph = elf_get_gnu_stack_phdr_ndx(ehdr, class, phnum);
        phdr = elf_get_program_header(ehdr, ndx_ph);
        if (write(packed, phdr.elf32, phdr_struct_size) != (ssize_t)phdr_struct_size)
  	  return (4);
        ,*writed += phdr_struct_size;
      }
    return (0);
  }

  	if (mprotect(map_addr, map_size, prot) < 0)
  	  return -1;
      }
      return 0;
  }

  // Generic ELF unpacking function that works for both main ELF and interpreter
  uint64_t load_elf_generic(void *ptr, uint64_t aslr_offset, uint64_t target_vaddr, int fd)
  {
      Elf64_Ehdr *elf = (Elf64_Ehdr *)ptr;
      Elf64_Phdr *elf_ph = (Elf64_Phdr *)((char *)ptr + elf->e_phoff);
      
      uint64_t min_vaddr, max_vaddr;
      
      // Calculate memory range needed
      calculate_elf_memory_range(elf_ph, elf->e_phnum, &min_vaddr, &max_vaddr);
      size_t total_size = max_vaddr - min_vaddr;
      
      // Reserve entire address space with PROT_NONE
      void *base_addr;
      base_addr = mmap((void *)(target_vaddr + aslr_offset), total_size, 
                          PROT_NONE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
      
      if (base_addr == MAP_FAILED)
          return (uint64_t)-1;
      
      // Load each segment
      for (int i = 0; i < elf->e_phnum; i++) {
          if (load_elf_segment(base_addr, ptr, &elf_ph[i], min_vaddr, fd) < 0) {
              munmap(base_addr, total_size);
              return (uint64_t)-1;
          }
      }
      
      uint64_t entrypoint = (uint64_t)base_addr + (elf->e_entry - min_vaddr);
      return entrypoint;
  }

  {
      Elf64_Ehdr *elf = (Elf64_Ehdr *)ptr;
      Elf64_Phdr *elf_ph = (Elf64_Phdr *)((char *)ptr + elf->e_phoff);
      uint64_t ndx_ph = 0;
      char *interp_str;
      int interp_fd;
      char interp_elfh[1024];
      
      // Find PT_INTERP segment
      while (ndx_ph < elf->e_phnum && elf_ph[ndx_ph].p_type != PT_INTERP)
          ndx_ph++;
      
      if (ndx_ph == elf->e_phnum)
          return 0;  // No interpreter found
      
      interp_str = (char *)ptr + elf_ph[ndx_ph].p_offset;
      interp_fd = open(interp_str, O_RDONLY);
      if (interp_fd < 0)
          return (uint64_t)-1;
      
      if (read(interp_fd, &interp_elfh[0], 1024) < 1024) {
          close(interp_fd);
          return (uint64_t)-1;
      }
      
      // Use the generic unpacker for the interpreter
      uint64_t result = load_elf_generic(&interp_elfh[0], aslr_offset, 0, interp_fd);
      
      close(interp_fd);
      return result;
  }

  size_t lz77_decompress_asm(const unsigned char *in, size_t in_size, unsigned char *out) {
    size_t orig_size;
    size_t pos = 0;
    uint16_t dl;
    size_t len, dist;
    const unsigned char *p = in + sizeof(orig_size);

    // Validate input
    if (!in || in_size < sizeof(orig_size) || !out)
      goto error;

    // Read original size
    orig_size = *((size_t *) in);
    if (!orig_size)
      goto error;

    // Decompress loop
    while ((p + 3 <= in + in_size) && pos < orig_size) {
      // Read distance/length as little-endian
      dl = p[0] | (p[1] << 8);
      dist = (dl >> 4) & 0xFFF;
      len = (dl & 0xF);
      if (len) len += 2;

      // Copy match data
      if (len && dist && dist <= pos) {
        if (len > orig_size - pos) 
  	len = orig_size - pos;
    	  
        unsigned char *dst = out + pos;
        unsigned char *src = out + (pos - dist);
    	  
        while (len--) 
  	,*dst++ = *src++;
        pos += (dst - (out + pos));  // Update pos
      }
      // Copy literal byte
      if (pos < orig_size)
        out[pos++] = p[2];
      p += 3;
    }
    // Check if we decompressed all data
    if (pos != orig_size)
      goto error;
    return 0;

   error:
    return -1;
  }
