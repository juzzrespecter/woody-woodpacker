  int get_file_size(int fd, off_t *file_size)
    {
    ,*file_size = lseek(fd, 0, SEEK_END);
    if (*file_size == ERROR_STDL)
    {
        perror("Failed to seek the end of file");
        return (FAIL_LSEEK);
    }
    if (lseek(fd, 0, SEEK_SET) == ERROR_STDL)
    {
        perror("Failed to seek the begin of file");
        return (FAIL_LSEEK);
    }
    return (SUCCESS);
  }

#+begin_src c :tangle srcs/signature_check.c
  #include "woody.h"

  t_file_type get_ftype(const char *file_content, const off_t *file_size)
  {

    if (ELF_MAGIC_SIZE < *file_size && !ft_memcmp(file_content, ELFMAG, SELFMAG))
      return (FT_ELF);
    if (DOS_SIGN_SIZE < *file_size && !ft_memcmp(file_content, DOS_SIGN, DOS_SIGN_SIZE))
       return (FT_PE);
    return (FT_UNKNOW);
  }
