    {
      t_elf_ph phdr;
      uint64_t i;
      size_t p_vaddr;
      size_t max_vaddr = 0;
      size_t max_align_size = 0;

      i = 0;
      while (++i < phnum) // Skip the first 
        {
          phdr = elf_get_program_header(ehdr, i);
          if (elf_get_program_header_type(&phdr, class) != PT_LOAD) // If not loadable ignore
            continue ;
          p_vaddr = elf_get_program_header_vaddr(&phdr, class);
          if (max_vaddr < p_vaddr)

              max_vaddr = p_vaddr;
              max_align_size = get_alignment_size(elf_get_program_header_align(&phdr, class), elf_get_program_header_memsz(&phdr, class));
            }
        }
      return  (max_vaddr + max_align_size);
    }

    void set_jmp_entrypoint(void *payload, uint64_t entrypoint, unsigned char class)
    {
      t_elf_addr u_entrypoint = (t_elf_addr){.b64 = entrypoint};
      size_t offset = 0;

      if (class == ELFCLASS64)
        {
          memcpy(payload, "\x48", 1); //Rex Prefix x64
          offset++;
        }
      memcpy(payload + offset, "\xB8", 1); //Mov eax in case preceding Rex is rax.
      offset++;
      if (class == ELFCLASS64)
        {
          memcpy(payload + offset, &u_entrypoint.b64, sizeof (u_entrypoint.b64));
          offset+= sizeof (u_entrypoint.b64);
        }
      else
        {
          memcpy(payload + offset, &u_entrypoint.b32, sizeof (u_entrypoint.b32));
          offset+= sizeof (u_entrypoint.b32);
        }
      memcpy(payload + offset, "\xFF\xE0", 2); 
    }

  {
    size_t written = 0;

    while (i < payload_size)
