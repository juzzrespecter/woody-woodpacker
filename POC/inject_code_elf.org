    uint64_t swapped;

  void create_jump_payload(uint64_t entry_point, char *payload) {
    // Assuming payload has enough space
    // Clear the payload
    memset(payload, 0, 64); // Adjust size as needed

    // Create a jump instruction
    // For a near jump, we use the opcode 0xE9
    payload[0] = 0xE9; // Opcode for JMP

    // Calculate the relative address
    // The address of the instruction following the jump
    uint64_t next_instruction_address = (uint64_t)(payload + 5); // 5 bytes for the jump instruction

    // Calculate the relative offset
    int32_t offset = (int32_t)(entry_point - next_instruction_address);

    // Copy the offset into the payload (little-endian)
    memcpy(payload + 1, &offset, sizeof(offset));
}
