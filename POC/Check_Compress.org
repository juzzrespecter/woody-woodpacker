  {
    uint16_t distance_len;
    unsigned char byte;

  #define COMPRESS_LZ77_MAX_MATCH_LEN 17
  #define COMPRESS_LZ77_MIN_MATCH_LEN 3
  #define COMPRESS_LZ77_SLIDING_WINDOW_SIZE 0x1000


  int lz77_compress(const unsigned char *data, size_t data_len, int fd) {
      if (!data || data_len == 0) return -1;

      // Write original size (8 bytes, little-endian)
      uint64_t original_size = data_len;
      if (write(fd, &original_size, sizeof(original_size)) != sizeof(original_size)) {
          return -1;
      }

      size_t i = 0;
      while (i < data_len) {
          size_t best_offset = 0;
          size_t best_length = 0;

          // Search in sliding window: [max(0, i - WINDOW_SIZE), i)
  	size_t window_start = 0;//(i > COMPRESS_LZ77_SLIDING_WINDOW_SIZE) * (i - COMPRESS_LZ77_SLIDING_WINDOW_SIZE);
  	if (i > COMPRESS_LZ77_SLIDING_WINDOW_SIZE) {
  	  window_start = i - COMPRESS_LZ77_SLIDING_WINDOW_SIZE;
  	}

          // Try all possible offsets (1 to window_size)
          for (size_t offset = 1; offset <= (i - window_start); offset++) {
              size_t match_length = 0;
  	    size_t max_possible = (data_len - i > COMPRESS_LZ77_MAX_MATCH_LEN) ? COMPRESS_LZ77_MAX_MATCH_LEN : (data_len - i);

  	    while (match_length < max_possible &&
  		   data[i - offset + match_length] == data[i + match_length]) {
  	      match_length++;
  	    }

              // Only consider matches of length >=3
              if (match_length >= COMPRESS_LZ77_MIN_MATCH_LEN && match_length > best_length) {
                  best_length = match_length;
                  best_offset = offset;
              }
          }

          uint8_t next_char = 0;
          if (i + best_length < data_len) {
              next_char = data[i + best_length];
          }

          t_lz77 token = {0};

          if (best_length >= COMPRESS_LZ77_MIN_MATCH_LEN) {
              // Valid match: store offset and len-2
              if (best_offset > 4095) {
                  fprintf(stderr, "Offset %zu too large\n", best_offset);
                  return -1;
              }
              size_t token_len = best_length - 2; // 1 to 15
              token.distance_len = (best_offset << 4) | (token_len & 0x0F);
              token.byte = next_char;
              i += best_length + 1; // Skip matched bytes + literal
          } else {
              // Literal only
              token.distance_len = 0; // offset=0, len=0
              token.byte = data[i];
              i += 1;
          }

          // Write token (3 bytes)
          if (write(fd, &token, sizeof(token)) != sizeof(token)) {
              return -1;
          }
  	printf("Token: distlen=0x%04x (offset=%zu, len=%zu), byte=0x%02x at i=%zu\n",
  	       token.distance_len,
  	       (size_t)(token.distance_len >> 4),
  	       (size_t)((token.distance_len & 0xF) == 0 ? 0 : ((token.distance_len & 0xF) + 2)),
  	       token.byte,
  	       i - (best_length >= 3 ? best_length + 1 : 1));
      }

      return 0;
  }

  size_t lz77_decompress_asm(const unsigned char *in, size_t in_size, unsigned char *out) {
      size_t orig_size;
      size_t pos = 0;
      uint16_t dl;
      size_t len, dist;
      const unsigned char *p = in + sizeof(orig_size);

      // Validate input
      if (!in || in_size < sizeof(orig_size) || !out)
          goto error;

      // Read original size
      orig_size = *((size_t *) in);
      if (!orig_size)
          goto error;

      // Decompress loop
      while ((p + 3 <= in + in_size) && pos < orig_size) {
        // Read distance/length as little-endian
        dl = p[0] | (p[1] << 8);
        dist = (dl >> 4) & 0xFFF;
        len = (dl & 0xF);
        if (len) len += 2;

        // Copy match data
        if (len && dist && dist <= pos) {
  	if (len > orig_size - pos) 
  	  len = orig_size - pos;
  	  
  	unsigned char *dst = out + pos;
  	unsigned char *src = out + (pos - dist);
  	  
  	while (len--) 
  	  ,*dst++ = *src++;
  	pos += (dst - (out + pos));  // Update pos
        }
        // Copy literal byte
        if (pos < orig_size)
  	out[pos++] = p[2];
        p += 3;
      }
      // Check if we decompressed all data
      if (pos != orig_size)
          goto error;
      return 0;

  error:
      return -1;
  }

  int lz77_decompress(const unsigned char *in, size_t in_size, int fd) {
      size_t orig_size;
      unsigned char *out;

      if (!in || in_size < sizeof(orig_size)) return -1;

      // Read original size
      orig_size = *(size_t*)in;
      if (!orig_size) return -2;

      out = __builtin_malloc(orig_size);
      if (!out) return -3;
      lz77_decompress_asm(in, in_size, out);
      write(fd, out, orig_size);
      __builtin_free(out);
      return orig_size ? 0 : -4;
  }

    //validation params
    if (argc != 3 || (toupper(argv[1][0]) != 'C' && toupper(argv[1][0]) != 'U'))
      {
        printf("usage: %s [C/U] [filename]\n", *argv);
        return (1);
      }
    //open file
    int fd = open(argv[2], O_RDONLY); 
    if (fd < 0)
      return (2);
    // get size file
    const off_t file_size = lseek(fd, 0, SEEK_END);
    if ((off_t) -1 == file_size || (off_t) -1 == lseek(fd, 0, SEEK_SET))
      {
        close(fd);
        return (3);
      }
    // save on memory all the file
    void *elf_mmap = mmap(NULL, file_size, PROT_READ, MAP_SHARED, fd, 0);
    close(fd);
    if (elf_mmap == (void *)-1)
      return (4);
    int i;
    i = 0;
    if (toupper(argv[1][0]) == 'C')
      {
        if (lz77_compress(elf_mmap, file_size, 2))
  	i = 5;
      }
    else
      {
        if (lz77_decompress(elf_mmap, file_size, 2))
  	i = 5;
      }
    munmap(elf_mmap, file_size);
    return (i);
  }

  int lz77_decompress(const char *compressed_data, size_t compressed_size)
  {
    size_t original_size;
    char *output = malloc(original_size);
    char window[COMPRESS_LZ77_SLIDING_WINDOW_SIZE] = {0};
    size_t window_start = 0, window_end = 0, window_size = 0, output_pos = 0;
    const char *current = compressed_data + sizeof(size_t);
    size_t length;
    t_lz77 *token;
    if (!compressed_data || compressed_size < sizeof(size_t))
      return (-1);

    memcpy(&original_size, compressed_data, sizeof(size_t));
      
    if (!output)
      return (-1);
      
    // Use your existing sliding window functions
    while ((current + sizeof(*token)) < (compressed_data + compressed_size) && output_pos < original_size) {
      token = (t_lz77*)current;
          
      // Unified processing
      length = 0;
      if (token->len > 0)
        length = (token->len + 2);
          
      // Copy referenced data using your get function
      for (size_t i = 0; i < length && output_pos < original_size; i++) {
        // For decompression, we want to read from (distance) positions back
        if (token->distance > 0 && token->distance <= window_size) {
  	size_t backward_idx = token->distance - 1; // 1 = most recent
  	char byte = get_sliding_window_char_from_end(window, window_end, window_size, backward_idx + i);
  	output[output_pos++] = byte;
        }
      }
          
      // Add next character
      if (output_pos < original_size)
        output[output_pos++] = token->byte;
          
      // Update window using your add function
      add_sliding_window(window, &window_start, &window_end, &window_size,
  		       output + (output_pos - length - 1), length + 1);
          
      current += sizeof(t_lz77);
    }
      
    if (output_pos == original_size) {
      write(1, output, output_pos);
      free(output);
      return (0);
    }
      
    free(output);
    return (-1);
  }
